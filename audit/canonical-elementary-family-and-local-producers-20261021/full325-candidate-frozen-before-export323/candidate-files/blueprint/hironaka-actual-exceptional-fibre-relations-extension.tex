\section{Actual exceptional-fibre flatness and all polynomial relations}

\begin{lemma}[Whole quotient base change of an actual flat ring map]
\label{hir:actual-flat-quotient-ring-base-change}
\lean{Hironaka.HironakaActualExceptionalFibreRelations.quotientMap_flat}
\leanok
If $P\to R$ is flat and $I\subseteq P$ is any whole ideal, the genuine
induced ring map $P/I\to R/IR$ is flat.
\end{lemma}
\begin{proof}
The actual quotient tensor isomorphism identifies $R/IR$ with
$R\otimes_P(P/I)$. Compose with tensor commutation to identify it with
$(P/I)\otimes_P R$. This is an isomorphism over $P/I$: its initially
$P$-linear formula respects the whole quotient scalar action since
$P\to P/I$ is surjective. Flatness of $R$ over $P$ gives flatness of
this actual base change over $P/I$. Transport it through the proved
whole quotient tensor isomorphism.
\end{proof}

\begin{theorem}[The genuine original exceptional-fibre polynomial evaluation is flat]
\label{hir:actual-original-exceptional-fibre-polynomial-flatness}
\uses{hir:actual-flat-quotient-ring-base-change,hir:actual-original-polynomial-analytic-flatness}
\lean{Hironaka.HironakaActualExceptionalFibreRelations.pointFibrePolynomialEval,Hironaka.HironakaActualExceptionalFibreRelations.pointFibrePolynomialEval_flat,Hironaka.HironakaActualExceptionalFibreRelations.pointPolynomialIdealImage_eq_coordinate_span}
\leanok
Let $P=\mathbb C[x_1,\ldots,x_n]$, let
$R=\mathcal O_{\mathbb C^n,0}$ be the actual original germ ring,
and let $e$ be an actual coordinate germ. Literal coordinate evaluation
induces a flat whole ring map $P/(x_e)\to R/(e)$, sending each actual
polynomial class to the class of its literal original evaluation.
Neither quotient-flatness nor a desired fibre identification is supplied.
\end{theorem}
\begin{proof}
The whole image of the principal polynomial ideal $(x_e)$ under actual
coordinate evaluation is the principal ideal of the actual coordinate
germ $e$, by the literal variable formula and the two whole span-image
directions. Actual entire polynomial evaluation is flat by the preceding
original producer. Apply whole quotient base change to this actual map
and principal ideal. Its induced quotient map is exactly the stated
numerator-preserving evaluation, by the genuine quotient-map formula.
\end{proof}

\begin{theorem}[Every genuine exceptional-fibre relation is generated by pivot-free polynomial syzygies]
\label{hir:actual-original-exceptional-fibre-polynomial-syzygies}
\uses{hir:actual-original-exceptional-fibre-polynomial-flatness}
\lean{Hironaka.HironakaActualExceptionalFibreRelations.exists_original_exceptional_polynomial_syzygies}
\leanok
Let $p_i$ be a finite family of literal complex polynomials independent
of $x_e$, and $A_i\in R$ actual holomorphic germs. If
$\sum_i p_iA_i\in(e)$ after literal original evaluation, there are
finitely many genuine pivot-free polynomial syzygies $c_{ij}$ and
actual holomorphic germs $B_j$ such that
\[
 A_i-\sum_j c_{ij}B_j\in(e),\qquad
 \sum_i p_ic_{ij}=0
\]
with literal evaluation in the first expression. The number,
polynomial syzygies and actual analytic multipliers are produced
internally. Every coefficient exponent of every $c_{ij}$ has pivot
component zero.
\end{theorem}
\begin{proof}
Pass the given actual relation to the whole quotient $R/(e)$. The
proved flatness over $P/(x_e)$ and the finite equational criterion
supply finitely many polynomial-quotient syzygies and actual
quotient multipliers expressing all the classes of $A_i$.
Choose polynomial representatives of the former and genuine analytic
representatives of the latter using the true quotient surjections.

Project a polynomial by setting $x_e=0$ and retaining every other
variable. This genuine algebra retraction leaves its class modulo
$(x_e)$ unchanged and leaves each given pivot-free $p_i$ unchanged.
Its image has no pivot variable: it is the rename of the polynomial
ring on precisely the complementary variables. Apply this retraction
to each represented polynomial-quotient syzygy. Its original sum lay
in $(x_e)$, which the retraction kills; hence the projected coefficients
satisfy the displayed whole polynomial syzygy, not merely a quotient
relation. Their quotient classes are unchanged, so the analytic
quotient multiplier identities become exactly the asserted whole
principal-ideal congruences. Every supported exponent has pivot
component zero by the stated complementary-variable image.
\end{proof}

\end{remark}
