\section{Whole polynomial-to-analytic flatness and actual polynomial syzygies}

\begin{lemma}[Flatness descends through a faithfully flat target]
\label{hir:actual-flatness-faithful-target-descent}
\lean{Hironaka.HironakaActualPolynomialAnalyticFlatness.flat_of_faithfully_flat_target}
\leanok
Let $P\to R\to S$ be a tower of commutative rings. If $S$ is flat as
a $P$-module and faithfully flat as an $R$-module, then $R$ is flat
as a $P$-module.
\end{lemma}
\begin{proof}
For a $P$-submodule inclusion $N\hookrightarrow M$, tensor first with
$R$ and then with $S$ over $R$. The canonical actual tensor
associativity and cancellation isomorphisms identify this second map
with $S\otimes_P N\to S\otimes_P M$, which is injective by the
assumed $P$-flatness of $S$. Faithful flatness over $R$ reflects
injectivity, so $R\otimes_P N\to R\otimes_P M$ is injective.
The tensor criterion for flatness proves the assertion.
\end{proof}

\begin{theorem}[The actual original polynomial evaluation is flat on the whole ring]
\label{hir:actual-original-polynomial-analytic-flatness}
\uses{hir:actual-flatness-faithful-target-descent,hir:actual-original-taylor-faithfully-flat,hir:actual-original-polynomial-low-coefficient-separation,hir:actual-original-formal-taylor-algebra-embedding}
\lean{Hironaka.HironakaActualPolynomialAnalyticFlatness.originalPolynomialEval_flat}
\leanok
For every $n$, literal evaluation of complex polynomials at the actual
original coordinate germs gives a flat whole-ring map
$\mathbb C[x_1,\ldots,x_n]\to\mathcal O_{\mathbb C^n,0}$.
No localization, flatness, Taylor-coefficient or relation certificate
is supplied.
\end{theorem}
\begin{proof}
Write $P=\mathbb C[x_1,\ldots,x_n]$, $R=\mathcal O_{\mathbb C^n,0}$
and $S=\mathbb C[[x_1,\ldots,x_n]]$. The actual formal series ring is
canonically the completion of $P$ at its whole variables ideal, and
this equivalence sends every literal polynomial to its completion
image. Since $P$ is Noetherian, its adic completion is flat over $P$;
transport through that actual equivalence proves $P$-flatness of $S$.
The genuine original Taylor map $R\to S$ is faithfully flat by its
actual original completion and whole ideal contraction. Its composition
with actual polynomial evaluation is exactly literal polynomial
inclusion in $S$: verify constants and variables by the genuine Taylor
coordinate identities, and then sums and products. Thus these are one
actual tower of ring maps, with no scalar-action replacement. Apply the
preceding faithful-target descent lemma to obtain flatness of the
entire actual polynomial evaluation map.
\end{proof}

\begin{theorem}[Every analytic relation on literal polynomials is generated by true polynomial syzygies]
\label{hir:actual-original-analytic-polynomial-relation-generation}
\uses{hir:actual-original-polynomial-analytic-flatness}
\lean{Hironaka.HironakaActualPolynomialAnalyticFlatness.actual_analytic_polynomial_relation_iff_polynomial_syzygies}
\leanok
Let $p_i$ be any finite family of actual complex polynomials and $A_i$
any family of original holomorphic germs. The whole relation
$\sum_i p_i A_i=0$ after actual coordinate evaluation holds if and only
if there are finitely many polynomial families $c_{ij}$ and actual
holomorphic germs $B_j$ such that
\[
 A_i=\sum_j c_{ij}B_j,\qquad \sum_i p_i c_{ij}=0
\]
with literal polynomial evaluation in the first equality and a whole
polynomial identity in the second. The number, polynomial syzygies and
analytic multipliers are produced internally from the actual relation.
\end{theorem}
\begin{proof}
Use the actual whole polynomial evaluation to give the analytic germ
ring its genuine polynomial module action. The preceding theorem proves
this actual module flat. The finite equational criterion for flatness
applied to the displayed true relation produces a finite trivialization:
polynomial coefficients $c_{ij}$, analytic $B_j$, the stated multiplier
identities, and exactly the polynomial syzygy identities. The actual
module action is multiplication by literal original polynomial
evaluation, so these are precisely the required whole analytic
identities. Conversely substitute the multiplier identities into the
original finite sum, distribute and interchange the two finite sums.
Each inner polynomial syzygy evaluates to zero; hence the original
whole analytic relation is zero.
\end{proof}

\begin{remark}
This result supplies genuine analytic generation of the relation module
of polynomial functions; it is not a flatness or prepared-tracking
certificate assumed at the application boundary. Rehomogenization of
exceptional-chart polynomial syzygies and the resulting normalized-ideal
exceptional regularity remain separate.
\end{remark}
