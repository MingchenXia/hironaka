/-!
# Wlodarczyk's lemma on tangent directions on an open set

Source: Wlodarczyk 2005 (`wlodarczyk05-hironaka`), §2.7, Lemma `le:Gi` (Giraud) parts (2) and (3):
for a function `u` of multiplicity one on an open set `U` and the blowup `π` along a centre `C` with
`C ∩ U ⊆ V(u)`, the controlled transform `u' = h⁻¹ · (u ∘ π)` (`h` a local equation of the
exceptional divisor) has multiplicity one, and `V(u')` is the strict transform of `V(u) ∩ U` over
`U`.  Blueprint `bcalc:hypersurface`, `bcalc:giraud-local` (chapter `blowup-calculus.tex`).

The global forms (`V(u) ⊇ C` a closed submanifold of the whole manifold) are
`BlowupFunctionTransform.mem_geoStrictTransform_iff_eq_zero` and
`GeneratorMultiplicityOne.exists_chartDeriv_ne_zero_controlledTransform`.  Here `V(u) ∩ U` is only
closed in `U`; the global statements are applied to the restriction `π⁻¹ U → U` of the blowup
(`BlowupRestrictOpen.isBlowupAlong_restrict`) and read back on `Y`.

* `exists_hypersurface_coords`, `vanishingIdeal_zeroSet_mem_iff`, `regularAtDim_zeroSet` -- the
  zero set of a function of multiplicity one on `U` is a hypersurface of `U` whose ideal is
  generated by the function (the inverse function theorem and Hadamard's lemma);
* `IsLocalEquation.restrict` -- local equations of the exceptional divisor restrict to `π⁻¹ U`;
* `vanishingIdeal_geoStrictTransform_mem_iff_of_open`,
  `mem_geoStrictTransform_iff_eq_zero_of_open`, `eventually_mem_geoStrictTransform_iff_of_open`
  -- `le:Gi` (3): `𝓘_{N'} = (u')` and `N' = V(u')` near every point of `π⁻¹ U` where `h` is defined,
  for `N = V(u) ∩ U`;
* `exists_chartDeriv_ne_zero_controlledTransform_of_open` -- `le:Gi` (2): `du' ≠ 0` on `N'`;
* `IsTangentDirection.giraud_hypotheses` -- a tangent direction `u` of `(𝔞, μ)` on `U`
  (TN23's `MaximalContact.IsTangentDirection`) and a centre in the cosupport satisfy the hypotheses
  of the above (Wlodarczyk: "in particular `supp(𝓘, μ) ∩ U ⊂ V(u)`").

Everything here is sorry-free, and outside the import closure of the two Hironaka statements.
-/

/-! ### The zero set of a function of multiplicity one -/

section Hypersurface

variable {X : Type*} [TopologicalSpace X] [ChartedSpace (ComplexModel n) X]
  [IsComplexManifold n X]

/-- **Coordinates in which a function of multiplicity one is the last coordinate.**  If `u` is
holomorphic at `x` with a nonvanishing chart derivative there, then `0 < n` and there is a map
`W : X → ℂⁿ`, holomorphic wherever `u` is (on the preferred chart at `x`), with injective
differential at `x` and last coordinate `u`. -/
theorem exists_hypersurface_coords {u : X → ℂ} {x : X} (hu : HolomorphicAt n u x)
    {e : ComplexChart n X} {v : ComplexModel n} (hxe : x ∈ e.val.source)
    (hd : chartDeriv e v u x ≠ 0) :
    ∃ (hn : 0 < n) (W : X → ComplexModel n),
      (∀ y ∈ (chartAt (ComplexModel n) x).source, HolomorphicAt n u y →
        ContMDiffAt 𝓘(ℂ, ComplexModel n) 𝓘(ℂ, ComplexModel n) ∞ W y) ∧
      Function.Injective (fderiv ℂ (fun ζ => W ((chartAt (ComplexModel n) x).symm ζ))
        (chartAt (ComplexModel n) x x)) ∧
      ∀ y, W y ⟨n - 1, Nat.sub_lt hn one_pos⟩ = u y := by
  classical
  set ex := complexChartAt (n := n) x with hex
  have hxx : x ∈ ex.val.source := mem_complexChartAt_source x
  -- some partial derivative in the preferred chart does not vanish
  have hj : ∃ j : Fin n, chartDeriv ex (EuclideanSpace.single j 1) u x ≠ 0 := by
    by_contra hall
    push Not at hall
    have h := (chartDeriv_eventuallyEq_sum ex e hxx hxe hu (v := v)).self_of_nhds
    simp only [hall, mul_zero, Finset.sum_const_zero] at h
    exact hd h
  obtain ⟨j, hj⟩ := hj
  have hn : 0 < n := Fin.pos j
  set last : Fin n := ⟨n - 1, Nat.sub_lt hn one_pos⟩ with hlast
  set τ : Equiv.Perm (Fin n) := Equiv.swap j last with hτ
  set Lsel := selMap τ (fun r => r ≠ last) with hLsel
  set W : X → ComplexModel n := fun y => Lsel (ex.val y) + u y • EuclideanSpace.single last 1
    with hW
  refine ⟨hn, W, fun y hy huy => ?_, ?_, fun y => ?_⟩
  · -- smoothness
    have h1 : ContMDiffAt 𝓘(ℂ, ComplexModel n) 𝓘(ℂ, ComplexModel n) ∞
        (fun y => Lsel (ex.val y)) y :=
      (Lsel.contMDiff.contMDiffAt).comp y
        ((contMDiffOn_chart (n := ∞)).contMDiffAt (ex.val.open_source.mem_nhds hy))
    have h2 : ContMDiffAt 𝓘(ℂ, ComplexModel n) 𝓘(ℂ, ℂ) ∞ u y :=
      contMDiffAt_of_eventually_mdifferentiableAt huy
    exact h1.add (h2.smul contMDiffAt_const)
  · -- the differential at `x`
    set a := fderiv ℂ (u ∘ ex.val.symm) (ex.val x) with ha
    have hud : DifferentiableAt ℂ (u ∘ ex.val.symm) (ex.val x) :=
      (eventually_differentiableAt_comp_symm ex hxx hu).self_of_nhds
    have hloc : (fun ζ => W (ex.val.symm ζ)) =ᶠ[𝓝 (ex.val x)]
        fun ζ => Lsel ζ + (u ∘ ex.val.symm) ζ • EuclideanSpace.single last (1 : ℂ) := by
      filter_upwards [ex.val.open_target.mem_nhds (ex.val.map_source hxx)] with ζ hζ
      simp only [hW, Function.comp_apply, ex.val.right_inv hζ]
    have hD : HasFDerivAt (fun ζ => W (ex.val.symm ζ))
        (Lsel + a.smulRight (EuclideanSpace.single last (1 : ℂ))) (ex.val x) := by
      refine HasFDerivAt.congr_of_eventuallyEq ?_ hloc
      exact Lsel.hasFDerivAt.add (hud.hasFDerivAt.smul_const _)
    show Function.Injective (fderiv ℂ (fun ζ => W (ex.val.symm ζ)) (ex.val x))
    rw [hD.fderiv, injective_iff_map_eq_zero]
    intro w hw
    have hcomp : ∀ r, Lsel w r + a w * (EuclideanSpace.single last (1 : ℂ)) r = 0 := by
      intro r
      have := congrArg (fun z : ComplexModel n => z r) hw
      simpa using this
    -- all coordinates but `j` vanish
    have hoff : ∀ k, k ≠ j → w k = 0 := by
      intro k hk
      have hτk : τ k ≠ last := by
        intro h
        apply hk
        have := congrArg τ h
        rwa [Equiv.swap_apply_self, hτ, Equiv.swap_apply_right] at this
      have := hcomp (τ k)
      simp only [hLsel, selMap_apply, hτk, ne_eq, not_false_eq_true, if_true, PiLp.ofLp_single, Pi.single_apply, if_false, mul_zero,
        add_zero] at this
      rwa [hτ, Equiv.swap_apply_self] at this
    have hwj : w = w j • EuclideanSpace.single j (1 : ℂ) := by
      have := eq_sum_single_of_support (T := {j}) (v := w) fun t ht => hoff t (by simpa using ht)
      simpa using this
    have hlastc := hcomp last
    simp only [hLsel, selMap_apply, ne_eq, not_true_eq_false, if_false, PiLp.ofLp_single,
      Pi.single_apply, if_true, mul_one, zero_add] at hlastc
    rw [hwj, map_smul, smul_eq_mul] at hlastc
    have hj' : a (EuclideanSpace.single j 1) ≠ 0 := hj
    have hwj0 : w j = 0 := (mul_eq_zero.mp hlastc).resolve_right hj'
    rw [hwj, hwj0, zero_smul]
  · -- the last coordinate
    show (Lsel (ex.val y) + u y • EuclideanSpace.single last (1 : ℂ)) last = u y
    simp [hLsel, selMap_apply]

/-- Membership of the last coordinate plane `ℂ^{n-1} × 0` is the vanishing of the last
coordinate. -/
theorem mem_plane_pred_iff (hn : 0 < n) (w : ComplexModel n) :
    w ∈ complexCoordinatePlane n (n - 1) ↔ w ⟨n - 1, Nat.sub_lt hn one_pos⟩ = 0 := by
  constructor
  · intro h
    exact h _ le_rfl
  · intro h i hi
    have : i = ⟨n - 1, Nat.sub_lt hn one_pos⟩ := Fin.ext (by show i.1 = n - 1; have := i.2; omega)
    rw [this]
    exact h

/-- **The ideal of the zero set of a function of multiplicity one is generated by the function**:
if `u` is holomorphic on the open set `U` with a nonvanishing chart derivative at each of its
zeros in `U`, then at every point of `U` the ideal of `V(u) ∩ U` is `(u)`. -/
theorem vanishingIdeal_zeroSet_mem_iff {U : Set X} (hU : IsOpen U) {u : X → ℂ}
    (hu : ∀ x ∈ U, HolomorphicAt n u x)
    (hu1 : ∀ x ∈ U, u x = 0 → ∃ (e : ComplexChart n X) (v : ComplexModel n),
      x ∈ e.val.source ∧ chartDeriv e v u x ≠ 0)
    {x : X} (hx : x ∈ U) (g : X → ℂ) :
    (vanishingIdeal (n := n) ({y | u y = 0} ∩ U)).mem x g ↔ IsInSpanAt (n := n) (· = u) x g := by
  by_cases hux : u x = 0
  · obtain ⟨e, v, hxe, hd⟩ := hu1 x hx hux
    obtain ⟨hn, W, hWs, hWinj, hWu⟩ := exists_hypersurface_coords (hu x hx) hxe hd
    have hO : IsOpen (U ∩ (chartAt (ComplexModel n) x).source) :=
      hU.inter (chartAt (ComplexModel n) x).open_source
    have hxO : x ∈ U ∩ (chartAt (ComplexModel n) x).source := ⟨hx, mem_chart_source _ x⟩
    have hplane : ∀ y, W y ∈ complexCoordinatePlane n (n - 1) ↔ u y = 0 := fun y => by
      rw [mem_plane_pred_iff hn, hWu]
    obtain ⟨U₀, -, hxU₀, -, hiff⟩ := exists_vanishingIdeal_mem_iff_planeCoords hO
      (fun z' hz' => hWs z' hz'.2 (hu z' hz'.1)) hxO hWinj ((hplane x).mpr hux)
      (A := {y | u y = 0} ∩ U) fun y hy => ⟨fun h => (hplane y).mpr h.1,
        fun h => ⟨(hplane y).mp h, hy.1⟩⟩
    rw [hiff x hxU₀ g]
    have hlast : planeCoords W (n - 1) ⟨n - 1, Nat.sub_lt hn one_pos⟩ = u := by
      funext y
      simp [planeCoords, hWu]
    constructor
    · refine fun hg => IsInSpanAt.trans (fun s hs => ?_) hg
      obtain ⟨i, rfl⟩ := hs
      by_cases hi : n - 1 ≤ i.1
      · have : i = ⟨n - 1, Nat.sub_lt hn one_pos⟩ := Fin.ext (by show i.1 = n - 1; have := i.2; omega)
        rw [this, hlast]
        exact isInSpanAt_of_eq_mul rfl (hu x hx) (holomorphicAt_const 1)
          (Eventually.of_forall fun _ => by simp)
      · simp only [planeCoords, hi, if_false]
        exact isInSpanAt_zero
    · exact fun hg => hg.mono fun s hs => ⟨⟨n - 1, Nat.sub_lt hn one_pos⟩, hlast.trans hs.symm⟩
  · -- off the zero set both sides consist of all holomorphic germs
    have hne : ∀ᶠ y in 𝓝 x, u y ≠ 0 := (hu x hx).continuousAt.eventually (isOpen_ne.mem_nhds hux)
    constructor
    · rintro ⟨hg, -⟩
      refine isInSpanAt_of_eq_mul rfl hg (hg.mul (holomorphicAt_inv (hu x hx) hux)) ?_
      filter_upwards [hne] with y hy
      field_simp
    · intro hg
      refine ⟨hg.holomorphicAt, ?_⟩
      filter_upwards [hne] with y hy hyA
      exact absurd hyA.1 hy

/-- **The zero set of a function of multiplicity one is a hypersurface**: regular of dimension
`n - 1` at each of its points in `U`. -/
theorem regularAtDim_zeroSet {U : Set X} (hU : IsOpen U) {u : X → ℂ}
    (hu : ∀ x ∈ U, HolomorphicAt n u x)
    (hu1 : ∀ x ∈ U, u x = 0 → ∃ (e : ComplexChart n X) (v : ComplexModel n),
      x ∈ e.val.source ∧ chartDeriv e v u x ≠ 0)
    {x : X} (hx : x ∈ U) (hux : u x = 0) :
    RegularAtDim (n := n) ({y | u y = 0} ∩ U) x (n - 1) := by
  obtain ⟨e, v, hxe, hd⟩ := hu1 x hx hux
  obtain ⟨hn, W, hWs, hWinj, hWu⟩ := exists_hypersurface_coords (hu x hx) hxe hd
  have hO : IsOpen (U ∩ (chartAt (ComplexModel n) x).source) :=
    hU.inter (chartAt (ComplexModel n) x).open_source
  have hxO : x ∈ U ∩ (chartAt (ComplexModel n) x).source := ⟨hx, mem_chart_source _ x⟩
  have hplane : ∀ y, W y ∈ complexCoordinatePlane n (n - 1) ↔ u y = 0 := fun y => by
    rw [mem_plane_pred_iff hn, hWu]
  obtain ⟨Θ, hΘt, hΘ0, hΘ⟩ :=
    exists_recentred_coords hO (fun z' hz' => hWs z' hz'.2 (hu z' hz'.1)) hxO hWinj
  refine ⟨Nat.sub_le n 1, ⟨hux, hx⟩, Θ, hΘt, hΘ0, fun {y} hy hyΘ => ?_⟩
  obtain ⟨hyO, hΘy⟩ := hΘ hy hyΘ
  rw [hΘy, sub_mem_plane_iff ((hplane x).mpr hux), hplane]
  exact ⟨fun h => h.1, fun h => ⟨h, hyO.1⟩⟩

/-- The zero set of a function continuous on the open set `U` is closed in `U`. -/
theorem isClosed_preimage_val_zeroSet {U : Opens X} {u : X → ℂ}
    (hu : ∀ x ∈ (U : Set X), HolomorphicAt n u x) :
    IsClosed ((↑) ⁻¹' ({y | u y = 0} ∩ (U : Set X)) : Set U) := by
  have hc : Continuous fun x : U => u x.1 :=
    continuous_iff_continuousAt.mpr fun x => ((hu x.1 x.2).continuousAt).comp
      continuous_subtype_val.continuousAt
  have : ((↑) ⁻¹' ({y | u y = 0} ∩ (U : Set X)) : Set U) = (fun x : U => u x.1) ⁻¹' {0} := by
    ext x
    simp
  rw [this]
  exact isClosed_singleton.preimage hc

end Hypersurface

/-! ### Restriction to an open subset -/

section Restrict

variable {X : Type u} [TopologicalSpace X] [ChartedSpace (ComplexModel n) X]
  [IsComplexManifold n X] {Y : Type u} [TopologicalSpace Y] [ChartedSpace (ComplexModel n) Y]
  [IsComplexManifold n Y] {C : Set X} {π : Y → X}

/-- Germs of the ideal of `A ∩ O` on the open subset `O` are the germs of the ideal of `A`. -/
theorem vanishingIdeal_comp_val_iff {Z : Type*} [TopologicalSpace Z]
    [ChartedSpace (ComplexModel n) Z] [IsComplexManifold n Z] (O : Opens Z) {A : Set Z} {x : O}
    {g : Z → ℂ} :
    (vanishingIdeal (n := n) ((↑) ⁻¹' A : Set O)).mem x (g ∘ Subtype.val) ↔
      (vanishingIdeal (n := n) A).mem x.1 g := by
  refine and_congr (holomorphicAt_comp_val_iff O) ?_
  exact (eventually_nhds_val_iff O (P := fun y => y ∈ A → g y = 0)).symm

/-- The span of one function on an open subset. -/
theorem isInSpanAt_eq_comp_val_iff {Z : Type*} [TopologicalSpace Z]
    [ChartedSpace (ComplexModel n) Z] [IsComplexManifold n Z] (O : Opens Z) {x : O} {f g : Z → ℂ} :
    IsInSpanAt (n := n) (· = f ∘ Subtype.val) x (g ∘ Subtype.val) ↔
      IsInSpanAt (n := n) (· = f) x.1 g := by
  have hS : (fun s' : O → ℂ => s' = f ∘ Subtype.val) =
      fun s' => ∃ s, s = f ∧ s' = s ∘ Subtype.val := by
    funext s'
    exact propext ⟨fun h => ⟨f, rfl, h⟩, fun ⟨s, hs, h⟩ => hs ▸ h⟩
  have hext : extendOpens O (g ∘ Subtype.val) =ᶠ[𝓝 x.1] g := by
    filter_upwards [O.isOpen.mem_nhds x.2] with y hy
    exact extendOpens_val O (g ∘ Subtype.val) ⟨y, hy⟩
  rw [hS, isInSpanAt_comp_val_iff O]
  exact ⟨fun h => isInSpanAt_congr h hext, fun h => isInSpanAt_congr h hext.symm⟩

/-- Strict transforms commute with restriction to an open subset (the same statement as
TN-CLASSICAL's `MarkedIdealRestrictOpen.geoStrictTransform_restrict`, kept here to avoid importing
that module's closure). -/
theorem geoStrictTransform_preimage_val (V : Opens X) (hV : IsOpen (π ⁻¹' (V : Set X)))
    (A : Set X) :
    geoStrictTransform (fun p : (⟨π ⁻¹' (V : Set X), hV⟩ : Opens Y) => (⟨π p.1, p.2⟩ : V))
        ((↑) ⁻¹' C : Set V) ((↑) ⁻¹' A) =
      (↑) ⁻¹' geoStrictTransform π C A :=
  ((Opens.isOpen ⟨π ⁻¹' (V : Set X), hV⟩).isOpenMap_subtype_val.preimage_closure_eq_closure_preimage
    continuous_subtype_val (π ⁻¹' (A \ C))).symm

/-- **Local equations of the exceptional divisor restrict to `π⁻¹ V`.** -/
theorem isLocalEquation_restrict {hπ : IsBlowupAlong n C π} {U' : Set Y} {h : Y → ℂ}
    (hh : IsLocalEquation hπ U' h) (V : Opens X) (hV : IsOpen (π ⁻¹' (V : Set X))) :
    IsLocalEquation (isBlowupAlong_restrict hπ V hV)
      ((↑) ⁻¹' U' : Set (⟨π ⁻¹' (V : Set X), hV⟩ : Opens Y)) (h ∘ Subtype.val) where
  isOpen := hh.isOpen.preimage continuous_subtype_val
  mdifferentiableOn := fun y hy =>
    ((mdifferentiableAt_comp_val_iff (⟨π ⁻¹' (V : Set X), hV⟩ : Opens Y)).mpr
      (hh.holomorphicAt hy).self_of_nhds).mdifferentiableWithinAt
  nonvanishing := fun W hW hWU ⟨w, hw⟩ => by
    obtain ⟨y, ⟨w', hw'W, rfl⟩, hy⟩ := hh.nonvanishing (Subtype.val '' W)
      ((Opens.isOpen ⟨π ⁻¹' (V : Set X), hV⟩).isOpenMap_subtype_val W hW)
      (by rintro _ ⟨w', hw', rfl⟩; exact hWU hw') ⟨w.1, w, hw, rfl⟩
    exact ⟨w', hw'W, hy⟩
  generates := fun y hy g => by
    rw [exceptionalDivisor_ideal, pullback_vanishingIdeal_opens V]
    have hS : (fun s : (⟨π ⁻¹' (V : Set X), hV⟩ : Opens Y) → ℂ =>
          ∃ a, (vanishingIdeal (n := n) C).mem
            ((Subtype.val ∘ fun p : (⟨π ⁻¹' (V : Set X), hV⟩ : Opens Y) => (⟨π p.1, p.2⟩ : V)) y) a ∧
            s = a ∘ (Subtype.val ∘ fun p : (⟨π ⁻¹' (V : Set X), hV⟩ : Opens Y) =>
              (⟨π p.1, p.2⟩ : V))) =
        fun s' => ∃ s, (∃ a, (vanishingIdeal (n := n) C).mem (π y.1) a ∧ s = a ∘ π) ∧
          s' = s ∘ Subtype.val := by
      funext s'
      exact propext ⟨fun ⟨a, ha, h⟩ => ⟨a ∘ π, ⟨a, ha, rfl⟩, h⟩,
        fun ⟨_, ⟨a, ha, h1⟩, h2⟩ => ⟨a, ha, by rw [h2, h1]; rfl⟩⟩
    have hT : (fun s' : (⟨π ⁻¹' (V : Set X), hV⟩ : Opens Y) → ℂ => s' = h ∘ Subtype.val) =
        fun s' => ∃ s, s = h ∧ s' = s ∘ Subtype.val := by
      funext s'
      exact propext ⟨fun h' => ⟨h, rfl, h'⟩, fun ⟨s, hs, h'⟩ => hs ▸ h'⟩
    rw [GermIdeal.mem_pullback_iff, hS, hT, isInSpanAt_comp_val_iff, isInSpanAt_comp_val_iff]
    have := hh.generates y.1 hy (extendOpens _ g)
    rw [exceptionalDivisor_ideal] at this
    exact this

end Restrict

/-! ### Wlodarczyk's lemma on tangent directions, parts (2) and (3), on an open set -/

section Giraud

variable {X : Type u} [TopologicalSpace X] [ChartedSpace (ComplexModel n) X]
  [IsComplexManifold n X] {Y : Type u} [TopologicalSpace Y] [ChartedSpace (ComplexModel n) Y]
  [IsComplexManifold n Y] {C : Set X} {π : Y → X} {hπ : IsBlowupAlong n C π} {U' : Set Y}
  {h : Y → ℂ}

/-- **`le:Gi` (3), ideal form, on an open set.**  Let `u` be holomorphic on the open set `U`, of
multiplicity one at its zeros in `U`, with `C ∩ U ⊆ V(u)`; let `h` be a local equation of the
exceptional divisor on `U'`, `z ∈ U' ∩ π⁻¹ U`, and `u ∘ π = h · u'` near `z` with `u'` holomorphic
at `z`.  Then the ideal of the strict transform `N'` of `N = V(u) ∩ U` is `(u')` at `z`. -/
theorem vanishingIdeal_geoStrictTransform_mem_iff_of_open (hh : IsLocalEquation hπ U' h)
    {U : Set X} (hU : IsOpen U) {u : X → ℂ} (hu : ∀ x ∈ U, HolomorphicAt n u x)
    (hu1 : ∀ x ∈ U, u x = 0 → ∃ (e : ComplexChart n X) (v : ComplexModel n),
      x ∈ e.val.source ∧ chartDeriv e v u x ≠ 0)
    (hCu : ∀ x ∈ C, x ∈ U → u x = 0) {z : Y} (hz : z ∈ U') (hzU : π z ∈ U) {u' : Y → ℂ}
    (hu' : HolomorphicAt n u' z) (heq : (u ∘ π) =ᶠ[𝓝 z] fun y => h y * u' y) (g : Y → ℂ) :
    (vanishingIdeal (n := n) (geoStrictTransform π C ({x | u x = 0} ∩ U))).mem z g ↔
      IsInSpanAt (n := n) (· = u') z g := by
  obtain ⟨V, rfl⟩ : ∃ V : Opens X, (V : Set X) = U := ⟨⟨U, hU⟩, rfl⟩
  have hV : IsOpen (π ⁻¹' (V : Set X)) := hU.preimage hπ.contMDiff.continuous
  have hr := isBlowupAlong_restrict hπ V hV
  have hNV : IsClosedSubmanifold n
      ((↑) ⁻¹' ({x | u x = 0} ∩ (V : Set X)) : Set V) :=
    ⟨isClosed_preimage_val_zeroSet (U := V) hu, fun x hx =>
      ⟨n - 1, regularAtDim_preimage_val_iff.mpr (regularAtDim_zeroSet hU hu hu1 x.2 hx.1)⟩⟩
  have hCNV : ((↑) ⁻¹' C : Set V) ⊆ (↑) ⁻¹' ({x | u x = 0} ∩ (V : Set X)) :=
    fun x hx => ⟨hCu x.1 hx x.2, x.2⟩
  have hhW := isLocalEquation_restrict hh V hV
  have huV : HolomorphicAt n (u ∘ Subtype.val)
      ((⟨π z, hzU⟩ : V)) :=
    (holomorphicAt_comp_val_iff V).mpr (hu _ hzU)
  have hgenV : ∀ f : V → ℂ,
      (vanishingIdeal (n := n) ((↑) ⁻¹' ({x | u x = 0} ∩ (V : Set X)) : Set V)).mem
        (⟨π z, hzU⟩ : V) f ↔ IsInSpanAt (n := n) (· = u ∘ Subtype.val) (⟨π z, hzU⟩ : V) f := by
    intro f
    have hf : f = extendOpens V f ∘ Subtype.val :=
      funext fun x => (extendOpens_val _ f x).symm
    rw [hf, vanishingIdeal_comp_val_iff, isInSpanAt_eq_comp_val_iff]
    exact vanishingIdeal_zeroSet_mem_iff hU hu hu1 hzU _
  have hu'W : HolomorphicAt n (u' ∘ Subtype.val)
      (⟨z, hzU⟩ : (⟨π ⁻¹' (V : Set X), hV⟩ : Opens Y)) :=
    (holomorphicAt_comp_val_iff (⟨π ⁻¹' (V : Set X), hV⟩ : Opens Y)).mpr hu'
  have heqW : ((u ∘ Subtype.val) ∘ fun p : (⟨π ⁻¹' (V : Set X), hV⟩ : Opens Y) =>
      (⟨π p.1, p.2⟩ : V)) =ᶠ[𝓝 ⟨z, hzU⟩]
        fun y => (h ∘ Subtype.val) y * (u' ∘ Subtype.val) y :=
    (eventually_nhds_val_iff (⟨π ⁻¹' (V : Set X), hV⟩ : Opens Y) (x := ⟨z, hzU⟩)
      (P := fun y => (u ∘ π) y = h y * u' y)).mp heq
  have key := vanishingIdeal_geoStrictTransform_mem_iff hhW (z := ⟨z, hzU⟩) hz hNV hCNV huV hgenV
    hu'W heqW (g ∘ Subtype.val)
  rw [geoStrictTransform_preimage_val V hV, vanishingIdeal_comp_val_iff,
    isInSpanAt_eq_comp_val_iff] at key
  exact key

/-- **`le:Gi` (3) on an open set: `V(u')` is the strict transform of `V(u) ∩ U`** at every point of
`U' ∩ π⁻¹ U` (hypotheses as in `vanishingIdeal_geoStrictTransform_mem_iff_of_open`). -/
theorem mem_geoStrictTransform_iff_eq_zero_of_open (hh : IsLocalEquation hπ U' h)
    {U : Set X} (hU : IsOpen U) {u : X → ℂ} (hu : ∀ x ∈ U, HolomorphicAt n u x)
    (hu1 : ∀ x ∈ U, u x = 0 → ∃ (e : ComplexChart n X) (v : ComplexModel n),
      x ∈ e.val.source ∧ chartDeriv e v u x ≠ 0)
    (hCu : ∀ x ∈ C, x ∈ U → u x = 0) {z : Y} (hz : z ∈ U') (hzU : π z ∈ U) {u' : Y → ℂ}
    (hu' : HolomorphicAt n u' z) (heq : (u ∘ π) =ᶠ[𝓝 z] fun y => h y * u' y) :
    z ∈ geoStrictTransform π C ({x | u x = 0} ∩ U) ↔ u' z = 0 :=
  mem_iff_eq_zero_of_span isClosed_closure hu'
    (vanishingIdeal_geoStrictTransform_mem_iff_of_open hh hU hu hu1 hCu hz hzU hu' heq)

/-- **`le:Gi` (3) on an open set, near a point**: `V(u') ∩ π⁻¹ U` and the strict transform of
`V(u) ∩ U` agree on a neighbourhood of every point of `U' ∩ π⁻¹ U`. -/
theorem eventually_mem_geoStrictTransform_iff_of_open (hh : IsLocalEquation hπ U' h)
    {U : Set X} (hU : IsOpen U) {u : X → ℂ} (hu : ∀ x ∈ U, HolomorphicAt n u x)
    (hu1 : ∀ x ∈ U, u x = 0 → ∃ (e : ComplexChart n X) (v : ComplexModel n),
      x ∈ e.val.source ∧ chartDeriv e v u x ≠ 0)
    (hCu : ∀ x ∈ C, x ∈ U → u x = 0) {z : Y} (hz : z ∈ U') (hzU : π z ∈ U) {u' : Y → ℂ}
    (hu' : HolomorphicAt n u' z) (heq : (u ∘ π) =ᶠ[𝓝 z] fun y => h y * u' y) :
    ∀ᶠ y in 𝓝 z, (y ∈ geoStrictTransform π C ({x | u x = 0} ∩ U) ↔ u' y = 0) := by
  have hzU' : π ⁻¹' U ∈ 𝓝 z := (hU.preimage hπ.contMDiff.continuous).mem_nhds hzU
  filter_upwards [hh.isOpen.mem_nhds hz, hzU', HolomorphicAt.eventually_holomorphicAt hu', heq.eventually_nhds]
    with y hy hyU hu'y heqy
  exact mem_geoStrictTransform_iff_eq_zero_of_open hh hU hu hu1 hCu hy hyU hu'y heqy

/-- **`le:Gi` (2) on an open set: the controlled transform has multiplicity one** on the strict
transform of `V(u) ∩ U` (hypotheses as in `vanishingIdeal_geoStrictTransform_mem_iff_of_open`). -/
theorem exists_chartDeriv_ne_zero_controlledTransform_of_open (hh : IsLocalEquation hπ U' h)
    {U : Set X} (hU : IsOpen U) {u : X → ℂ} (hu : ∀ x ∈ U, HolomorphicAt n u x)
    (hu1 : ∀ x ∈ U, u x = 0 → ∃ (e : ComplexChart n X) (v : ComplexModel n),
      x ∈ e.val.source ∧ chartDeriv e v u x ≠ 0)
    (hCu : ∀ x ∈ C, x ∈ U → u x = 0) {z : Y} (hz : z ∈ U') (hzU : π z ∈ U) {u' : Y → ℂ}
    (hu' : HolomorphicAt n u' z) (heq : (u ∘ π) =ᶠ[𝓝 z] fun y => h y * u' y)
    (hzN : z ∈ geoStrictTransform π C ({x | u x = 0} ∩ U)) :
    ∃ (e : ComplexChart n Y) (v : ComplexModel n), z ∈ e.val.source ∧ chartDeriv e v u' z ≠ 0 := by
  obtain ⟨V, rfl⟩ : ∃ V : Opens X, (V : Set X) = U := ⟨⟨U, hU⟩, rfl⟩
  have hV : IsOpen (π ⁻¹' (V : Set X)) := hU.preimage hπ.contMDiff.continuous
  have hr := isBlowupAlong_restrict hπ V hV
  have hNV : IsClosed ((↑) ⁻¹' ({x | u x = 0} ∩ (V : Set X)) : Set V) :=
    isClosed_preimage_val_zeroSet (U := V) hu
  have hNk : ∀ x ∈ ((↑) ⁻¹' ({x | u x = 0} ∩ (V : Set X)) : Set V),
      RegularAtDim (n := n) ((↑) ⁻¹' ({x | u x = 0} ∩ (V : Set X)) : Set V) x (n - 1) :=
    fun x hx => regularAtDim_preimage_val_iff.mpr (regularAtDim_zeroSet hU hu hu1 x.2 hx.1)
  have hCNV : ((↑) ⁻¹' C : Set V) ⊆ (↑) ⁻¹' ({x | u x = 0} ∩ (V : Set X)) :=
    fun x hx => ⟨hCu x.1 hx x.2, x.2⟩
  have hzW : (⟨z, hzU⟩ : (⟨π ⁻¹' (V : Set X), hV⟩ : Opens Y)) ∈
      geoStrictTransform (fun p : (⟨π ⁻¹' (V : Set X), hV⟩ : Opens Y) => (⟨π p.1, p.2⟩ : V))
        ((↑) ⁻¹' C) ((↑) ⁻¹' ({x | u x = 0} ∩ (V : Set X))) := by
    rw [geoStrictTransform_preimage_val V hV]
    exact hzN
  have hreg := regularAtDim_geoStrictTransform hr hNV hNk hCNV hzW
  rw [geoStrictTransform_preimage_val V hV] at hreg
  have hreg' := regularAtDim_preimage_val_iff.mp hreg
  -- the dimension is positive
  have hπz : u (π z) = 0 := by
    have h1 := geoStrictTransform_subset_preimage hr.contMDiff.continuous hNV hzW
    exact h1.1
  obtain ⟨e, v, hxe, hd⟩ := hu1 (π z) hzU hπz
  obtain ⟨hn, -⟩ := exists_hypersurface_coords (hu _ hzU) hxe hd
  exact exists_chartDeriv_ne_zero_of_vanishingIdeal_eq_span hreg' (by omega) hu'
    (vanishingIdeal_geoStrictTransform_mem_iff_of_open hh hU hu hu1 hCu hz hzU hu' heq)

/-- **A tangent direction satisfies the hypotheses of the local lemma on tangent directions**:
for a tangent direction `u` of `(𝔞, μ)` on `U` (`μ ≥ 1`, germs of `𝔞` holomorphic) and a centre
`C` in the cosupport of `(𝔞, μ)`, `U` is open, `u` is holomorphic on `U` of multiplicity one at its
zeros there, and `C ∩ U ⊆ V(u)`. -/
theorem IsTangentDirection.giraud_hypotheses {M : Type*} [TopologicalSpace M]
    [ChartedSpace (ComplexModel n) M] [IsComplexManifold n M] {𝔞 : GermIdeal n M}
    (h𝔞 : ∀ x g, 𝔞.mem x g → HolomorphicAt n g x) {μ : ℕ} (hμ : 1 ≤ μ) {U : Set M} {u : M → ℂ}
    (hu : MaximalContact.IsTangentDirection 𝔞 μ U u) {C : Set M}
    (hC : C ⊆ GermIdealOrder.cosupport n univ 𝔞 μ) :
    IsOpen U ∧ (∀ x ∈ U, HolomorphicAt n u x) ∧
      (∀ x ∈ U, u x = 0 → ∃ (e : ComplexChart n M) (v : ComplexModel n),
        x ∈ e.val.source ∧ chartDeriv e v u x ≠ 0) ∧
      ∀ x ∈ C, x ∈ U → u x = 0 :=
  ⟨hu.isOpen, fun x hx => holomorphicAt_of_mem_iterDerivIdeal h𝔞 (μ - 1) x u (hu.mem x hx),
    hu.multiplicity_one, fun _ hxC hxU =>
      MaximalContact.IsTangentDirection.cosupport_inter_subset h𝔞 hμ hu ⟨hC hxC, hxU⟩⟩

end Giraud
