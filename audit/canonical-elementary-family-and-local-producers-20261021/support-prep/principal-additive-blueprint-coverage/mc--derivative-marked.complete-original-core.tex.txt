\begin{definition}[Derivatives of a marked ideal]
\label{mc:derivative-marked}
\uses{hir:marked-ideal,hir:derivative-ideal,hir:cosupport}
\source{wlodarczyk05-hironaka}
\lean{TranscendentalNakamaye.Pending.DerivativeIdealTower.derivMarked,TranscendentalNakamaye.Pending.DerivativeIdealTower.derivMarked_N,TranscendentalNakamaye.Pending.DerivativeIdealTower.derivMarked_E,TranscendentalNakamaye.Pending.DerivativeIdealTower.derivMarked_ideal,TranscendentalNakamaye.Pending.DerivativeIdealTower.derivMarked_d,TranscendentalNakamaye.Pending.DerivativeIdealTower.cosupport_subset_of_le,TranscendentalNakamaye.Pending.DerivativeIdealTower.controlledTransform_le_of_le,TranscendentalNakamaye.Pending.DerivativeIdealTower.cosupp_subset_cosupp_derivMarked}
\leanok
For a marked ideal \(\underline{\mathcal I}=(M,N,E,\mathfrak a,d)\) and \(j\in\mathbb N\), \(\mathcal D^j(\underline{\mathcal I})=(M,N,E,\mathcal D^j(\mathfrak a),d-j)\).
If \(\mathfrak b\subset\mathfrak c\), then \(\operatorname{cosupp}(\mathfrak c,d)\subset\operatorname{cosupp}(\mathfrak b,d)\) and
\(\sigma^{\mathrm c}(\mathfrak b,\mu)\subset\sigma^{\mathrm c}(\mathfrak c,\mu)\). For \(N=M\) and \(j\le d\),
\(\operatorname{cosupp}\underline{\mathcal I}\subset\operatorname{cosupp}\mathcal D^j(\underline{\mathcal I})\).
\end{definition}
