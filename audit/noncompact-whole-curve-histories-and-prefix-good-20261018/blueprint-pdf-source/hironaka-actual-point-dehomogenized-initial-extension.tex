\section{The whole exceptional restriction is the actual dehomogenized initial polynomial}

Fix a genuine point chart with exceptional coordinate $e$ and actual

finite operations in $\mathbb C[x_1,\ldots,x_n]$.

\begin{definition}[The actual point-chart polynomial dehomogenization]
\label{hir:actual-point-polynomial-dehomogenization}
\lean{Hironaka.HironakaActualPointDehomogenizedInitial.actualPointDehomogenization}
\leanok
Define the actual complex polynomial algebra homomorphism
$\delta_{e,b}$ by $x_e\mapsto1$ and $x_i\mapsto b_i+x_i$ for $i\ne e$.
\end{definition}

\begin{lemma}[Whole polynomial inclusion retains literal chart evaluation]
\label{hir:actual-point-polynomial-dehomogenization-inclusion}
\uses{hir:actual-point-polynomial-dehomogenization}
\lean{Hironaka.HironakaActualPointDehomogenizedInitial.actualPointDehomogenization_coe}
\leanok
The actual formal inclusion of $\delta_{e,b}P$ is precisely the finite
polynomial evaluation at $1$ in coordinate $e$ and $b_i+x_i$ otherwise.
\end{lemma}
\begin{proof}
Both maps preserve the original complex constants, addition and
multiplication. On each actual variable they have the displayed identical
value. Polynomial induction proves equality for every finite original
polynomial, hence equality of the entire formal inclusions.
\end{proof}

\begin{theorem}[The actual point blowup factors every homogeneous polynomial]
\label{hir:actual-point-blowup-homogeneous-polynomial-factor}
\uses{hir:actual-point-polynomial-dehomogenization-inclusion,hir:actual-formal-blowup-coordinate-power-image}
\lean{Hironaka.HironakaActualPointDehomogenizedInitial.actualFormalBlowup_homogeneous_polynomial}
\leanok
For every actual homogeneous polynomial $P$ of degree $d$, the existing
actual blowup substitution satisfies
\[
 \pi^*P=x_e^d\delta_{e,b}P
\]
as equality of entire actual formal series.
\end{theorem}
\begin{proof}
Every actual point-blowup coordinate is $x_e$ times its genuine
dehomogenized coordinate. The original formal substitution on a finite
polynomial is its actual polynomial evaluation. Expand that polynomial
as its finite support sum of coefficient monomials. Each monomial has
total degree $d$, so its evaluated product contains exactly the common
factor $x_e^d$. Factor this from the finite sum and use the preceding
actual polynomial-inclusion identity. No substitution into an arbitrary
nonconvergent series at a nonzero point occurs.
\end{proof}

\begin{lemma}[The actual whole degree-$d$ polynomial is genuinely homogeneous]
\label{hir:actual-point-initial-polynomial-homogeneous}
\uses{hir:actual-point-initial-polynomial-whole-inclusion}
\lean{Hironaka.HironakaActualPointDehomogenizedInitial.actualPointHomogeneousPolynomial_isHomogeneous}
\leanok
For every entire actual series $F$, its actual polynomial $P_d(F)$ is
homogeneous of degree $d$, including when it is zero.
\end{lemma}
\begin{proof}
The whole polynomial inclusion equals the actual homogeneous component.
Every nonzero polynomial coefficient is therefore a nonzero coefficient of
that same whole component, whose literal homogeneity forces total degree
$d$. This is exactly the actual polynomial homogeneity predicate.
\end{proof}

\begin{theorem}[Every whole exceptional restriction is the genuine finite dehomogenized initial polynomial]

\begin{proof}

\end{proof}
