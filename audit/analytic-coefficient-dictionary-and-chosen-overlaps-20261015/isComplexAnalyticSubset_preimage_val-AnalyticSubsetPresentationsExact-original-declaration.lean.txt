    (hA : IsComplexAnalyticSubset (n := n) A) (U : Opens M) :
    IsComplexAnalyticSubset (n := n) ((↑) ⁻¹' A : Set U) :=
  fun x => ⟨presentationPreimageVal x (hA x).some⟩
