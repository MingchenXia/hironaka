/-!
# The flat case of the Remmert--Stein theorem

Blueprint `rem:remmert-stein-flat` (Demailly, *Complex analytic and differential geometry*, II.8.7).
Assume `(R_{m-1})`.  Let `Ω ⊆ ℂⁿ` be open, `L = ℂᵖ × {0}`, and `E ⊆ Ω \ L` analytic in `Ω \ L` with
`dim E ≤ m` and `dim_y E ≥ p + 1` at every point.  Then `closure E` is analytic at every point of
`Ω` (`remmertStein_flat`).

* `Wset`: the analytic set `W ⊆ Q` cut out by the extended canonical defining functions
  `P_{λ_i}(π z; λ_i(κ z - κ a))`; on the good base it agrees with `E` (`mem_Wset_iff`), and its
  fibres over `B'` are finite (`finite_Wset_fibre`: roots of monic polynomials, Vandermonde).
* `closure_inter_eq`: `closure E ∩ Q` is the closure of the `s`-dimensional regular stratum of
  `closure (W ∩ X₀)` (the blueprint's `W₁ˢ`), hence analytic in `Q`.
* `remmertStein_flat_pure`: the case of pure dimension `s = p + t`; `remmertStein_flat`: the general
  case, by the dimension strata of `E`.

Sorry-free.
-/

variable {n p t : ℕ} {Λ : ComplexModel n ≃L[ℂ] ComplexModel (p + t) × ComplexModel (n - (p + t))}
  {a : ComplexModel n} {δ ρ : ℝ}

/-- The `i`-th equation of `W`. -/
def eqn (Λ : ComplexModel n ≃L[ℂ] ComplexModel (p + t) × ComplexModel (n - (p + t)))
    (a : ComplexModel n) (q : ℕ) (F : ℕ → ℕ → ComplexModel (p + t) → ℂ) (i : ℕ)
    (z : ComplexModel n) : ℂ :=
  form (n - (p + t)) 0 i ((Λ z).2 - (Λ a).2) ^ q +
    ∑ j ∈ Finset.range q, F i j (Λ z).1 * form (n - (p + t)) 0 i ((Λ z).2 - (Λ a).2) ^ j

/-- The set `W ⊆ Q`. -/
def Wset (Λ : ComplexModel n ≃L[ℂ] ComplexModel (p + t) × ComplexModel (n - (p + t)))
    (a : ComplexModel n) (δ ρ : ℝ) (q : ℕ) (F : ℕ → ℕ → ComplexModel (p + t) → ℂ) :
    Set (ComplexModel n) :=
  {z | z ∈ polydisc Λ a δ ρ ∧ ∀ i ∈ Finset.range (q * (n - (p + t)) + 1), eqn Λ a q F i z = 0}

theorem differentiableOn_eqn {q : ℕ} {F : ℕ → ℕ → ComplexModel (p + t) → ℂ}
    (hFd : ∀ i j, DifferentiableOn ℂ (F i j) (ball (Λ a).1 δ)) (i : ℕ) :
    DifferentiableOn ℂ (eqn Λ a q F i) (polydisc Λ a δ ρ) := by
  have hform : Differentiable ℂ fun z : ComplexModel n => form (n - (p + t)) 0 i ((Λ z).2 - (Λ a).2) :=
    (form (n - (p + t)) 0 i).differentiable.comp
      ((((ContinuousLinearMap.snd ℂ _ _).comp Λ.toContinuousLinearMap).differentiable).sub_const
        (Λ a).2)
  have hfst : Differentiable ℂ fun z : ComplexModel n => (Λ z).1 :=
    ((ContinuousLinearMap.fst ℂ _ _).comp Λ.toContinuousLinearMap).differentiable
  show DifferentiableOn ℂ (fun z => form (n - (p + t)) 0 i ((Λ z).2 - (Λ a).2) ^ q +
    ∑ j ∈ Finset.range q, F i j (Λ z).1 * form (n - (p + t)) 0 i ((Λ z).2 - (Λ a).2) ^ j) _
  refine (hform.pow q).differentiableOn.add (DifferentiableOn.fun_sum (u := Finset.range q)
    fun j _ => ?_)
  refine DifferentiableOn.mul ?_ (hform.pow j).differentiableOn
  exact (hFd i j).comp hfst.differentiableOn fun z hz => by
    rw [mem_ball, dist_eq_norm]; exact hz.1

/-- `W` has a presentation at every point of `Q`. -/
theorem nonempty_presentation_Wset {q : ℕ} {F : ℕ → ℕ → ComplexModel (p + t) → ℂ}
    (hFd : ∀ i j, DifferentiableOn ℂ (F i j) (ball (Λ a).1 δ)) {x : ComplexModel n}
    (hx : x ∈ polydisc Λ a δ ρ) :
    Nonempty (ComplexAnalyticPresentationAt (n := n) (M := ComplexModel n) (Wset Λ a δ ρ q F) x) :=
  nonempty_presentation_of_eqOn isOpen_polydisc hx
    (fun k : Fin (q * (n - (p + t)) + 1) => eqn Λ a q F k)
    (fun k => differentiableOn_eqn hFd k) fun y hy => by
      simp only [Wset, mem_ofPred_eq, Finset.mem_range, hy, true_and]
      exact ⟨fun h k => h k k.2, fun h i hi => h ⟨i, hi⟩⟩

/-- The canonical polynomial of a finite fibre is monic of degree the cardinality. -/
theorem fibrePoly_monic {s d : ℕ} {E : Set (ComplexModel s × ComplexModel d)}
    (l : ComplexModel d →L[ℂ] ℂ) {w : ComplexModel s} (hfin : (fibreSet E w).Finite) :
    (fibrePoly E l w).Monic ∧ (fibrePoly E l w).natDegree = (fibreSet E w).ncard := by
  classical
  rw [fibrePoly_eq l hfin]
  refine ⟨monic_prod_of_monic _ _ fun x _ => monic_X_sub_C _, ?_⟩
  rw [natDegree_prod_of_monic _ _ fun x _ => monic_X_sub_C _]
  simp [Set.ncard_eq_toFinset_card _ hfin]

/-- Evaluation of a monic polynomial of degree `q` through its coefficients. -/
theorem eval_monic_eq {P : ℂ[X]} (hP : P.Monic) {q : ℕ} (hq : P.natDegree = q) (x : ℂ) :
    P.eval x = x ^ q + ∑ j ∈ Finset.range q, P.coeff j * x ^ j := by
  conv_lhs => rw [hP.as_sum]
  simp [hq, eval_finsetSum]

/-- **`W` agrees with `E` over the good base.** -/
theorem mem_Wset_iff {E : Set (ComplexModel n)} {B₀ : Set (ComplexModel (p + t))} {q : ℕ}
    (hq : ∀ w ∈ B₀, (fibreSet (coverSet Λ a δ ρ E B₀) w).Finite ∧
      (fibreSet (coverSet Λ a δ ρ E B₀) w).ncard = q)
    {F : ℕ → ℕ → ComplexModel (p + t) → ℂ}
    (hFeq : ∀ i j, EqOn (F i j)
      (fun w => (fibrePoly (coverSet Λ a δ ρ E B₀) (form (n - (p + t)) 0 i) w).coeff j) B₀)
    {z : ComplexModel n} (hzQ : z ∈ polydisc Λ a δ ρ) (hzB : (Λ z).1 ∈ B₀) :
    z ∈ Wset Λ a δ ρ q F ↔ z ∈ E := by
  set Ê := coverSet Λ a δ ρ E B₀ with hÊ
  set v := (Λ z).2 - (Λ a).2 with hv
  have hmemÊ : ((Λ z).1, v) ∈ Ê ↔ z ∈ E := by
    constructor
    · rintro ⟨y, ⟨⟨hyE, -⟩, -⟩, hyeq⟩
      simp only [Prod.mk.injEq, hv, sub_left_inj] at hyeq
      have : Λ y = Λ z := Prod.ext hyeq.1 hyeq.2
      rwa [Λ.injective this] at hyE
    · intro hzE
      exact ⟨z, ⟨⟨hzE, hzQ⟩, hzB⟩, rfl⟩
  rw [← hmemÊ, exists_forms_mem_iff hq hzB v]
  obtain ⟨hfin, hcard⟩ := hq _ hzB
  simp only [Wset, mem_ofPred_eq, hzQ, true_and]
  refine forall₂_congr fun i _ => ?_
  obtain ⟨hmon, hdeg⟩ := fibrePoly_monic (form (n - (p + t)) 0 i) hfin
  rw [eval_monic_eq hmon (hdeg.trans hcard), eqn]
  have : ∀ j ∈ Finset.range q, F i j (Λ z).1 =
      (fibrePoly Ê (form (n - (p + t)) 0 i) (Λ z).1).coeff j := fun j _ => hFeq i j hzB
  rw [Finset.sum_congr rfl fun j hj => by rw [this j hj]]

/-- **The fibres of `π` on `W` are finite.** -/
theorem finite_Wset_fibre {q : ℕ} {F : ℕ → ℕ → ComplexModel (p + t) → ℂ}
    (w : ComplexModel (p + t)) : (Wset Λ a δ ρ q F ∩ {z | (Λ z).1 = w}).Finite := by
  classical
  -- the polynomials
  set P : ℕ → ℂ[X] := fun i => X ^ q + ∑ j ∈ Finset.range q, C (F i j w) * X ^ j with hP
  have hPeval : ∀ i x, (P i).eval x = x ^ q + ∑ j ∈ Finset.range q, F i j w * x ^ j := by
    intro i x
    simp [hP, eval_finsetSum]
  have hPne : ∀ i, P i ≠ 0 := by
    intro i h
    have := congrArg (fun P => P.coeff q) h
    simp only [hP, coeff_add, coeff_X_pow, if_true, finsetSum_coeff, coeff_C_mul_X_pow,
      coeff_zero] at this
    rw [Finset.sum_eq_zero fun j hj => if_neg (by simp at hj; omega)] at this
    simp at this
  have heqn : ∀ z, (Λ z).1 = w → ∀ i, eqn Λ a q F i z =
      (P i).eval (form (n - (p + t)) 0 i ((Λ z).2 - (Λ a).2)) := by
    intro z hz i
    rw [hPeval, eqn, hz]
  rcases Nat.eq_zero_or_pos q with hq0 | hqpos
  · refine Set.finite_empty.subset fun z hz => ?_
    have := hz.1.2 0 (by simp)
    rw [heqn z hz.2, hPeval] at this
    simp [hq0] at this
  -- the root sets and the Vandermonde map
  set R : Fin (q * (n - (p + t)) + 1) → Set ℂ := fun i => {x | (P i).IsRoot x} with hR
  have hRfin : ∀ i, (R i).Finite := fun i => finite_setOfPred_isRoot (hPne i)
  set Fv : ComplexModel (n - (p + t)) → (Fin (q * (n - (p + t)) + 1) → ℂ) := fun v i => form (n - (p + t)) 0 i v with hFv
  have hFvinj : Function.Injective Fv := by
    intro v v' h
    have h0 : v - v' = 0 := by
      refine eq_zero_of_form_eq_zero (n := n - (p + t)) (d := 0) (I := Finset.range (q * (n - (p + t)) + 1)) ?_
        (Subsingleton.elim _ _) fun i hi => ?_
      · simp only [Finset.card_range, Nat.sub_zero]
        nlinarith
      · have := congrFun h ⟨i, Finset.mem_range.mp hi⟩
        simp only [hFv] at this
        rw [map_sub, this, sub_self]
    exact sub_eq_zero.mp h0
  have hpi : (Fv ⁻¹' Set.pi univ R).Finite :=
    (Set.Finite.pi fun i => hRfin i).preimage hFvinj.injOn
  refine (hpi.image fun v => Λ.symm (w, v + (Λ a).2)).subset fun z hz => ?_
  refine ⟨(Λ z).2 - (Λ a).2, fun i _ => ?_, ?_⟩
  · have := hz.1.2 i (Finset.mem_range.mpr i.2)
    rw [heqn z hz.2] at this
    exact this
  · simp only [sub_add_cancel]
    rw [← hz.2]
    simp

end W

/-- The coordinate `p`-plane is closed. -/
theorem isClosed_complexCoordinatePlane (n p : ℕ) : IsClosed (complexCoordinatePlane n p) := by
  have : complexCoordinatePlane n p = ⋂ i : Fin n, {z : ComplexModel n | p ≤ i.1 → z i = 0} := by
    ext z
    simp [complexCoordinatePlane]
  rw [this]
  refine isClosed_iInter fun i => ?_
  by_cases h : p ≤ i.1
  · simpa [h] using isClosed_eq (EuclideanSpace.proj i : ComplexModel n →L[ℂ] ℂ).continuous
      continuous_const
  · simp [h]

/-- Closures in an open subset are traces of closures. -/
theorem closure_opens_eq {M : Type*} [TopologicalSpace M] (U : Opens M) (T : Set U) :
    closure T = (↑) ⁻¹' closure ((↑) '' T : Set M) := by
  ext x
  exact closure_subtype

/-- **Sets over the bad set have dimension `≤ s - 1`**, when their fibres are finite. -/
theorem dimLE_over_badSet {Sig₀ : Set (coverTarget Λ a δ)}
    (hSiga : IsComplexAnalyticSubset (n := p + t) Sig₀)
    (hSigdim : DimLE (n := p + t) Sig₀ (p + t - 1)) {S : Set (ComplexModel n)}
    (hfin : ∀ w ∈ ball (Λ a).1 δ \ complexCoordinatePlane (p + t) p,
      (S ∩ polydisc Λ a δ ρ ∩ {z | (Λ z).1 = w}).Finite) :
    DimLE (n := n) (S ∩ (coverSource Λ a δ ρ : Set (ComplexModel n)) ∩
      {z | (Λ z).1 ∈ ((↑) '' Sig₀ : Set (ComplexModel (p + t)))}) (p + t - 1) := by
  set X₀ := coverSource Λ a δ ρ with hX₀
  set G := coverMap Λ a δ ρ with hG
  set T : Set X₀ := (↑) ⁻¹' (S ∩ (X₀ : Set (ComplexModel n)) ∩
    {z | (Λ z).1 ∈ ((↑) '' Sig₀ : Set (ComplexModel (p + t)))}) with hT
  have hTdim : DimLE (n := n) T (p + t - 1) := by
    refine FiniteMapDimension.dimLE_of_discrete_fibres (contMDiff_coverMap.mdifferentiable (by simp))
      (fun y _ => ?_) hSiga ?_ hSigdim
    · refine not_accPt_of_finite ?_ y
      have hy := y.2
      refine ((hfin (Λ y).1 ⟨by rw [mem_ball, dist_eq_norm]; exact hy.1.1, hy.2⟩).preimage
        Subtype.val_injective.injOn).subset fun z hz => ?_
      refine ⟨⟨hz.1.1.1, z.2.1⟩, ?_⟩
      have : G z = G y := hz.2
      exact congrArg Subtype.val this
    · rintro _ ⟨y, hy, rfl⟩
      obtain ⟨σ, hσ, hσy⟩ := hy.2
      have : coverMap Λ a δ ρ y = σ := Subtype.ext hσy.symm
      rw [this]
      exact hσ
  rw [hT, ← dimLE_image_val_iff] at hTdim
  convert hTdim using 1
  ext z
  constructor
  · intro hz
    exact ⟨⟨z, hz.1.2⟩, hz, rfl⟩
  · rintro ⟨y, hy, rfl⟩
    exact hy

/-- **The flat case of Remmert--Stein, pure dimension** (`rem:remmert-stein-flat`).  Under
`(R_{m-1})`, if `E ⊆ Ω \ L` is analytic in `Ω \ L` of pure dimension `p + t` (`t ≥ 1`,
`p + t ≤ m`), then `closure E` has a presentation at every point of `L ∩ Ω`. -/
theorem remmertStein_flat_pure {m : ℕ} (hR : ProperImageStatement.{0, 0} (m - 1)) {n p t : ℕ}
    (ht : 1 ≤ t) (hsn : p + t ≤ n) (hsm : p + t ≤ m) {Ω E : Set (ComplexModel n)}
    (hΩ : IsOpen Ω) (hEΩ : E ⊆ Ω) (hEL : Disjoint E (complexCoordinatePlane n p))
    (hEpres : ∀ z ∈ Ω, z ∉ complexCoordinatePlane n p →
      Nonempty (ComplexAnalyticPresentationAt (n := n) E z))
    (hpure : IsPureDim (n := n) E (p + t)) {a : ComplexModel n} (haΩ : a ∈ Ω)
    (haL : a ∈ complexCoordinatePlane n p) :
    Nonempty (ComplexAnalyticPresentationAt (n := n) (M := ComplexModel n) (closure E) a) := by
  classical
  have hLc := isClosed_complexCoordinatePlane n p
  set U : Opens (ComplexModel n) := ⟨Ω \ complexCoordinatePlane n p, hΩ.sdiff hLc⟩ with hU
  have hEU : E ⊆ U := fun z hz => ⟨hEΩ hz, fun h => disjoint_left.mp hEL hz h⟩
  have hEUa : IsComplexAnalyticSubset (n := n) ((↑) ⁻¹' E : Set U) :=
    isComplexAnalyticSubset_preimage_val_of_forall fun z hz => hEpres z hz.1 hz.2
  obtain ⟨Λ, hπL, hLinj, hTA, hD0⟩ := exists_flat_coordinates ht hsn (U := U)
    (fun z hz => hz.2) hEU hEUa hpure.dimLE haL
  have hcl : closure E ∩ Ω ⊆ E ∪ complexCoordinatePlane n p := by
    rintro z ⟨hzc, hzΩ⟩
    by_cases hzL : z ∈ complexCoordinatePlane n p
    · exact Or.inr hzL
    · exact Or.inl (closure_inter_subset_of_forall (hΩ.sdiff hLc)
        (fun x hx => hEpres x hx.1 hx.2) ⟨hzc, hzΩ, hzL⟩)
  -- regular points of `E` are dense
  have hreg_near : ∀ y ∈ E, ∀ V ∈ 𝓝 y, ∃ y' ∈ V, RegularAtDim (n := n) E y' (p + t) := by
    intro y hy V hV
    have hV' : ((↑) ⁻¹' V : Set U) ∈ 𝓝 (⟨y, hEU hy⟩ : U) :=
      continuous_subtype_val.continuousAt.preimage_mem_nhds hV
    obtain ⟨y', hy'V, k, hy'⟩ := exists_regularAtDim_mem_nhds hEUa (x := ⟨y, hEU hy⟩) hy hV'
    have hk := regularAtDim_preimage_val_iff.mp hy'
    exact ⟨y', hy'V, hpure _ _ hk ▸ hk⟩
  -- the central fibre is countable
  have hD : (E ∩ {z | (Λ z).1 = (Λ a).1}).Countable := by
    have hfst : ContMDiff 𝓘(ℂ, ComplexModel n) 𝓘(ℂ, ComplexModel (p + t)) ∞
        fun z : ComplexModel n => (Λ z).1 :=
      ((ContinuousLinearMap.fst ℂ _ _).contDiff.comp Λ.contDiff).contMDiff
    have hlev : IsComplexAnalyticSubset (n := n) (M := ComplexModel n)
        {z | (Λ z).1 = (Λ a).1} :=
      AnalyticSubsetPreimage.isComplexAnalyticSubset_preimage hfst
        (FibreDimensionLowerBound.isComplexAnalyticSubset_singleton (Λ a).1)
    set D' : Set U := (↑) ⁻¹' (E ∩ {z | (Λ z).1 = (Λ a).1}) with hD'
    have hD'a : IsComplexAnalyticSubset (n := n) D' :=
      hEUa.inter (isComplexAnalyticSubset_preimage_val hlev U)
    have hD'dim : DimLE (n := n) D' 0 := dimLE_preimage_val hD0 U
    have hdisc : IsDiscrete D' := isDiscrete_iff_nhdsNE.mpr fun y hy => by
      have h := not_accPt_of_locDimLE_zero hD'a hy (hD'dim.locDimLE y)
      rw [AccPt, not_neBot] at h
      exact h
    have hc : D'.Countable :=
      (HereditarilyLindelofSpace.isLindelof D').countable_of_isDiscrete hdisc
    refine (hc.image ((↑) : U → ComplexModel n)).mono fun z hz => ?_
    exact ⟨⟨z, hEU hz.1⟩, hz, rfl⟩
  obtain ⟨δ, hδ, ρ, hρ, hQΩ, hsph⟩ := exists_radii hΩ hLc Λ haΩ hcl hLinj hD
  obtain ⟨Sig₀, hSiga, hSigdim, hgood⟩ :=
    exists_badSet hR hsm hEL hEpres hpure hπL hcl hQΩ hsph
  have hB₀o : IsOpen (goodBase Λ a δ Sig₀) := isOpen_goodBase hSiga.isClosed
  have hB₀c := isPreconnected_base ht hδ hSiga hSigdim
  have hB₀sub : goodBase Λ a δ Sig₀ ⊆ ball (Λ a).1 δ := fun w hw => hw.1.1
  -- the covering
  have hπLK : ∀ K ⊆ goodBase Λ a δ Sig₀, ∀ l ∈ complexCoordinatePlane n p, (Λ l).1 ∉ K :=
    fun K hK l hl h => (hK h).1.2 (hπL l hl)
  have hprop : ∀ K ⊆ goodBase Λ a δ Sig₀, IsCompact K →
      IsCompact (coverSet Λ a δ ρ E (goodBase Λ a δ Sig₀) ∩ Prod.fst ⁻¹' K) := by
    intro K hK hKc
    have h := isCompact_inter_preimage hcl hQΩ hsph hKc (hK.trans hB₀sub) (hπLK K hK)
    convert h.image (show Continuous fun z => ((Λ z).1, (Λ z).2 - (Λ a).2) from
      (continuous_fst.comp Λ.continuous).prodMk
        ((continuous_snd.comp Λ.continuous).sub continuous_const)) using 1
    ext ⟨w, v⟩
    constructor
    · rintro ⟨⟨z, ⟨⟨hzE, hzQ⟩, -⟩, hzq⟩, hwK⟩
      refine ⟨z, ⟨⟨hzE, hzQ⟩, ?_⟩, hzq⟩
      simp only [Prod.mk.injEq] at hzq
      simp only [mem_preimage] at hwK
      rw [mem_ofPred_eq, hzq.1]
      exact hwK
    · rintro ⟨z, ⟨⟨hzE, hzQ⟩, hzK⟩, hzq⟩
      refine ⟨⟨z, ⟨⟨hzE, hzQ⟩, hK hzK⟩, hzq⟩, ?_⟩
      simp only [Prod.mk.injEq] at hzq
      simp only [mem_preimage, ← hzq.1]
      exact hzK
  have hgraph : ∀ q ∈ coverSet Λ a δ ρ E (goodBase Λ a δ Sig₀), ∃ V ∈ 𝓝 q,
      ∃ U' : Set (ComplexModel (p + t)), IsOpen U' ∧
        ∃ g : ComplexModel (p + t) → ComplexModel (n - (p + t)), DifferentiableOn ℂ g U' ∧
          coverSet Λ a δ ρ E (goodBase Λ a δ Sig₀) ∩ V = (fun w => (w, g w)) '' U' := by
    rintro _ ⟨z, ⟨⟨hzE, hzQ⟩, hzB⟩, rfl⟩
    exact exists_local_graph hsn hSiga.isClosed hgood hzE hzQ hzB
  have hEsub : coverSet Λ a δ ρ E (goodBase Λ a δ Sig₀) ⊆
      goodBase Λ a δ Sig₀ ×ˢ ball (0 : ComplexModel (n - (p + t))) ρ := by
    rintro _ ⟨z, ⟨⟨-, hzQ⟩, hzB⟩, rfl⟩
    exact ⟨hzB, by rw [mem_ball, dist_zero_right]; exact hzQ.2⟩
  obtain ⟨q, hq⟩ := exists_card_eq hB₀o hB₀c hprop hgraph
  -- the extended canonical coefficients
  have hext : ∀ i j : ℕ, ∃ Fij : ComplexModel (p + t) → ℂ,
      DifferentiableOn ℂ Fij (ball (Λ a).1 δ) ∧
      EqOn Fij (fun w => (fibrePoly (coverSet Λ a δ ρ E (goodBase Λ a δ Sig₀))
        (form (n - (p + t)) 0 i) w).coeff j) (goodBase Λ a δ Sig₀) := by
    intro i j
    refine exists_extension ht hSiga hSigdim
      (differentiableOn_coeff_fibrePoly hB₀o hprop hgraph _ j)
      (C := (1 + ‖form (n - (p + t)) 0 i‖ * ρ) ^ q) fun w hw => ?_
    have := norm_coeff_fibrePoly_le hEsub hρ.le (form (n - (p + t)) 0 i) (hq w hw).1 j
    rwa [(hq w hw).2] at this
  choose F hFd hFeq using hext
  -- `W`
  set Q := polydisc Λ a δ ρ with hQ
  set W := Wset Λ a δ ρ q F with hW
  set X₀ : Set (ComplexModel n) := (coverSource Λ a δ ρ : Set (ComplexModel n)) with hX₀
  set B₀ := goodBase Λ a δ Sig₀ with hB₀
  have hQo : IsOpen Q := isOpen_polydisc
  have hX₀Q : X₀ ⊆ Q := fun z hz => hz.1
  have hX₀o : IsOpen X₀ := (coverSource Λ a δ ρ).isOpen
  have hWQ : W ⊆ Q := fun z hz => hz.1
  have hWpres : ∀ x ∈ Q, Nonempty (ComplexAnalyticPresentationAt (n := n) (M := ComplexModel n)
      W x) := fun x hx => nonempty_presentation_Wset hFd hx
  have hWcl : closure W ∩ Q ⊆ W := closure_inter_subset_of_forall hQo hWpres
  have hW1 : ∀ z ∈ Q, (Λ z).1 ∈ B₀ → (z ∈ W ↔ z ∈ E) := fun z hzQ hzB =>
    mem_Wset_iff hq hFeq hzQ hzB
  have hdimE := dimLE_over_badSet (ρ := ρ) (S := E) hSiga hSigdim fun w hw =>
    finite_fibre hcl hEL hEpres hQΩ hsph hw.1 fun l hl h => hw.2 (h ▸ hπL l hl)
  have hdimW := dimLE_over_badSet (ρ := ρ) (S := W) hSiga hSigdim fun w _ =>
    (finite_Wset_fibre (Λ := Λ) (a := a) (δ := δ) (ρ := ρ) (q := q) (F := F) w).subset
      fun z hz => ⟨hz.1.1, hz.2⟩
  have hB₀X : ∀ z ∈ Q, (Λ z).1 ∈ B₀ → z ∈ X₀ := fun z hzQ hzB => ⟨hzQ, hzB.1.2⟩
  have hX₀B : ∀ z ∈ X₀, (Λ z).1 ∉ ((↑) '' Sig₀ : Set (ComplexModel (p + t))) → (Λ z).1 ∈ B₀ :=
    fun z hz hzS => ⟨⟨by rw [mem_ball, dist_eq_norm]; exact hz.1.1, hz.2⟩, hzS⟩
  -- the two analytic sets of the blueprint
  set Wa := closure (W ∩ X₀) with hWa
  set Ws := closure (PureDimension.regularStratum (n := n) (M := ComplexModel n) Wa
    (· = p + t) ∩ Q) with hWs
  set Qo : Opens (ComplexModel n) := ⟨Q, hQo⟩ with hQodef
  have hWaQ : IsComplexAnalyticSubset (n := n) ((↑) ⁻¹' Wa : Set Qo) := by
    have hWQa : IsComplexAnalyticSubset (n := n) ((↑) ⁻¹' W : Set Qo) :=
      isComplexAnalyticSubset_preimage_val_of_forall fun x hx => hWpres x hx
    have hfst : ContMDiff 𝓘(ℂ, ComplexModel n) 𝓘(ℂ, ComplexModel (p + t)) ∞
        fun z : Qo => (Λ z).1 :=
      (((ContinuousLinearMap.fst ℂ _ _).contDiff.comp Λ.contDiff).contMDiff).comp
        contMDiff_subtype_val
    have hNa : IsComplexAnalyticSubset (n := n)
        ((fun z : Qo => (Λ z).1) ⁻¹' complexCoordinatePlane (p + t) p) :=
      AnalyticSubsetPreimage.isComplexAnalyticSubset_preimage hfst
        (isComplexAnalyticSubset_plane p t)
    have h := ClosureDifferenceAnalytic.isComplexAnalyticSubset_closure_diff hWQa hNa
    rw [closure_opens_eq] at h
    convert h using 3
    ext z
    constructor
    · rintro ⟨hzW, hzX⟩
      exact ⟨⟨z, hzX.1⟩, ⟨hzW, hzX.2⟩, rfl⟩
    · rintro ⟨y, ⟨hyW, hyN⟩, rfl⟩
      exact ⟨hyW, y.2, hyN⟩
  have hWsQ : IsComplexAnalyticSubset (n := n) ((↑) ⁻¹' Ws : Set Qo) := by
    have h := PureDimension.isComplexAnalyticSubset_closure_regularStratum hWaQ (· = p + t)
    rw [closure_opens_eq] at h
    convert h using 3
    ext z
    constructor
    · rintro ⟨hz, hzQ⟩
      refine ⟨⟨z, hzQ⟩, ?_, rfl⟩
      rw [PureDimension.mem_regularStratum] at hz ⊢
      obtain ⟨j, hj, hzj⟩ := hz
      exact ⟨j, hj, regularAtDim_preimage_val_iff.mpr hzj⟩
    · rintro ⟨y, hy, rfl⟩
      rw [PureDimension.mem_regularStratum] at hy
      obtain ⟨j, hj, hyj⟩ := hy
      exact ⟨PureDimension.mem_regularStratum.mpr ⟨j, hj, regularAtDim_preimage_val_iff.mp hyj⟩,
        y.2⟩
  -- the good part `E ∩ O₀`
  set O₀ := Q ∩ {z | (Λ z).1 ∈ B₀} with hO₀
  have hO₀o : IsOpen O₀ := hQo.inter (hB₀o.preimage (continuous_fst.comp Λ.continuous))
  have hEWa : E ∩ O₀ = Wa ∩ O₀ := by
    ext z
    constructor
    · rintro ⟨hzE, hzO⟩
      exact ⟨subset_closure ⟨(hW1 z hzO.1 hzO.2).mpr hzE, hB₀X z hzO.1 hzO.2⟩, hzO⟩
    · rintro ⟨hzW, hzO⟩
      have : z ∈ W := hWcl ⟨closure_mono inter_subset_left hzW, hzO.1⟩
      exact ⟨(hW1 z hzO.1 hzO.2).mp this, hzO⟩
  have hsub1 : E ∩ O₀ ⊆ PureDimension.regularStratum (n := n) (M := ComplexModel n) Wa
      (· = p + t) ∩ Q := by
    rintro z ⟨hzE, hzO⟩
    have hzX := hB₀X z hzO.1 hzO.2
    have hzSig : coverMap Λ a δ ρ ⟨z, hzX⟩ ∉ Sig₀ := fun h => hzO.2.2 ⟨_, h, rfl⟩
    have hreg := regularAtDim_preimage_val_iff.mp (hgood ⟨z, hzX⟩ hzE hzSig).1
    refine ⟨PureDimension.mem_regularStratum.mpr ⟨p + t, rfl, ?_⟩, hzO.1⟩
    exact RegularPointDimension.regularAtDim_of_inter_eq hO₀o hzO hEWa hreg
  -- density of the good part in `E ∩ Q`
  have hdens : E ∩ Q ⊆ closure (E ∩ O₀) := by
    rintro z ⟨hzE, hzQ⟩
    rw [_root_.mem_closure_iff]
    intro O hO hzO
    -- step a: a point of `E` near `z` off `π⁻¹ N`
    obtain ⟨y₂, ⟨hy₂O, hy₂Q⟩, hy₂E, hy₂N⟩ : ∃ y₂ ∈ O ∩ Q, y₂ ∈ E ∧
        (Λ y₂).1 ∉ complexCoordinatePlane (p + t) p := by
      by_contra hcon
      push Not at hcon
      obtain ⟨y₁, hy₁, hy₁r⟩ := hreg_near z hzE (O ∩ Q) ((hO.inter hQo).mem_nhds ⟨hzO, hzQ⟩)
      have heq : E ∩ (O ∩ Q) = (E ∩ {z | (Λ z).1 ∈ complexCoordinatePlane (p + t) p}) ∩ (O ∩ Q) := by
        ext y
        constructor
        · rintro ⟨hyE, hyOQ⟩
          exact ⟨⟨hyE, hcon y hyOQ hyE⟩, hyOQ⟩
        · rintro ⟨⟨hyE, -⟩, hyOQ⟩
          exact ⟨hyE, hyOQ⟩
      have := hTA y₁ (p + t) (RegularPointDimension.regularAtDim_of_inter_eq (hO.inter hQo) hy₁ heq
        hy₁r)
      omega
    -- step b: a point of `E` near `y₂` off `π⁻¹ Σ₀`
    have hy₂X : y₂ ∈ X₀ := ⟨hy₂Q, hy₂N⟩
    by_contra hcon
    obtain ⟨y₃, hy₃, hy₃r⟩ := hreg_near y₂ hy₂E (O ∩ X₀) ((hO.inter hX₀o).mem_nhds ⟨hy₂O, hy₂X⟩)
    have heq : E ∩ (O ∩ X₀) = (E ∩ X₀ ∩ {z | (Λ z).1 ∈ ((↑) '' Sig₀ :
        Set (ComplexModel (p + t)))}) ∩ (O ∩ X₀) := by
      ext y
      constructor
      · rintro ⟨hyE, hyO, hyX⟩
        refine ⟨⟨⟨hyE, hyX⟩, ?_⟩, hyO, hyX⟩
        by_contra hyS
        exact hcon ⟨y, hyO, hyE, hyX.1, hX₀B y hyX hyS⟩
      · rintro ⟨⟨⟨hyE, -⟩, -⟩, hyOX⟩
        exact ⟨hyE, hyOX⟩
    have := hdimE y₃ (p + t) (RegularPointDimension.regularAtDim_of_inter_eq (hO.inter hX₀o) hy₃
      heq hy₃r)
    omega
  -- regular points of `Wa` of dimension `s` lie in the closure of the good part
  have hA : ∀ z ∈ X₀, RegularAtDim (n := n) Wa z (p + t) → z ∈ closure (E ∩ O₀) := by
    intro z hzX hz
    have hWaX : Wa ∩ X₀ = W ∩ X₀ := by
      ext y
      constructor
      · rintro ⟨hyW, hyX⟩
        exact ⟨hWcl ⟨closure_mono inter_subset_left hyW, hyX.1⟩, hyX⟩
      · rintro ⟨hyW, hyX⟩
        exact ⟨subset_closure ⟨hyW, hyX⟩, hyX⟩
    have hzW := RegularPointDimension.regularAtDim_of_inter_eq hX₀o hzX hWaX hz
    rw [_root_.mem_closure_iff]
    intro O hO hzO
    by_contra hcon
    have heq : W ∩ (O ∩ X₀) = (W ∩ X₀ ∩ {z | (Λ z).1 ∈ ((↑) '' Sig₀ :
        Set (ComplexModel (p + t)))}) ∩ (O ∩ X₀) := by
      ext y
      constructor
      · rintro ⟨hyW, hyO, hyX⟩
        refine ⟨⟨⟨hyW, hyX⟩, ?_⟩, hyO, hyX⟩
        by_contra hyS
        have hyB := hX₀B y hyX hyS
        exact hcon ⟨y, hyO, (hW1 y hyX.1 hyB).mp hyW, hyX.1, hyB⟩
      · rintro ⟨⟨⟨hyW, -⟩, -⟩, hyOX⟩
        exact ⟨hyW, hyOX⟩
    have := hdimW z (p + t) (RegularPointDimension.regularAtDim_of_inter_eq (hO.inter hX₀o)
      ⟨hzO, hzX⟩ heq hzW)
    omega
  have hB : ∀ z ∈ Q, RegularAtDim (n := n) Wa z (p + t) → z ∈ closure (E ∩ O₀) := by
    intro z hzQ hz
    by_cases hzN : (Λ z).1 ∈ complexCoordinatePlane (p + t) p
    · have hev := RegularPatchNeighborhood.eventually_regularAtDim hz
      obtain ⟨V, hVsub, hVo, hzV⟩ := _root_.mem_nhds_iff.mp hev
      rw [_root_.mem_closure_iff]
      intro O hO hzO
      have hzWa : z ∈ closure (W ∩ X₀) := hz.2.1
      obtain ⟨y, ⟨⟨hyO, hyQ⟩, hyV⟩, hyW, hyX⟩ := _root_.mem_closure_iff.mp hzWa (O ∩ Q ∩ V)
        ((hO.inter hQo).inter hVo) ⟨⟨hzO, hzQ⟩, hzV⟩
      have hyreg := hVsub hyV (subset_closure ⟨hyW, hyX⟩)
      have hyc := hA y hyX hyreg
      obtain ⟨w, hwO, hwE⟩ := _root_.mem_closure_iff.mp hyc (O ∩ Q ∩ V)
        ((hO.inter hQo).inter hVo) ⟨⟨hyO, hyQ⟩, hyV⟩
      exact ⟨w, hwO.1.1, hwE⟩
    · exact hA z ⟨hzQ, hzN⟩ hz
  -- `closure E ∩ Q = Ws ∩ Q`
  have hkey : closure E ∩ Q = Ws ∩ Q := by
    ext z
    constructor
    · rintro ⟨hzc, hzQ⟩
      refine ⟨?_, hzQ⟩
      have h1 : z ∈ closure (E ∩ Q) := hQo.closure_inter ⟨hzc, hzQ⟩
      have h2 : closure (E ∩ Q) ⊆ closure (E ∩ O₀) := by
        rw [← closure_closure (s := E ∩ O₀)]
        exact closure_mono hdens
      exact closure_mono hsub1 (h2 h1)
    · rintro ⟨hzW, hzQ⟩
      refine ⟨?_, hzQ⟩
      have h1 : PureDimension.regularStratum (n := n) (M := ComplexModel n) Wa (· = p + t) ∩ Q ⊆
          closure (E ∩ O₀) := by
        rintro y ⟨hy, hyQ⟩
        rw [PureDimension.mem_regularStratum] at hy
        obtain ⟨j, rfl, hyj⟩ := hy
        exact hB y hyQ hyj
      have h2 : Ws ⊆ closure (E ∩ O₀) := by
        rw [hWs, ← closure_closure (s := E ∩ O₀)]
        exact closure_mono h1
      exact closure_mono inter_subset_left (h2 hzW)
  obtain ⟨P⟩ := nonempty_presentation_of_preimage_val (U := Qo) hWsQ
    (mem_polydisc_self (Λ := Λ) (a := a) hδ hρ)
  exact nonempty_presentation_of_inter_eq hQo (mem_polydisc_self hδ hρ) hkey P

end PureTheorem
