/-- Every formal relation on an ACTUAL finite original family is generated by internally supplied analytic syzygies. -/
theorem actual_formal_relation_iff_original_syzygies (n r : ℕ)
    (G : Fin r → HolGerm (ComplexModel (0+n)))
    (A : Fin r → MvPowerSeries (Fin n) ℂ) :
    (∑ i, originalTaylorAlgHom n (G i)*A i)=0 ↔
      ∃ (k : ℕ) (c : Fin r → Fin k → HolGerm (ComplexModel (0+n)))
        (B : Fin k → MvPowerSeries (Fin n) ℂ),
        (∀ i,A i=∑ j,originalTaylorAlgHom n (c i j)*B j) ∧
        ∀ j,∑ i,G i*c i j=0 := by
  let R := HolGerm (ComplexModel (0+n))
  let S := MvPowerSeries (Fin n) ℂ
  let : Algebra R S := (originalTaylorAlgHom n).toRingHom.toAlgebra
  have : Module.Flat R S := originalTaylorAlgHom_flat n
  constructor
  · intro h
    have hs : (∑ i,G i • A i)=0 := by
      simpa only [Algebra.smul_def,RingHom.algebraMap_toAlgebra,
        AlgHom.toRingHom_eq_coe,AlgHom.coe_toRingHom] using h
    obtain ⟨k,c,B,hc,hG⟩ := (Module.Flat.iff_forall_isTrivialRelation (R := R) (M := S)).mp
      ‹Module.Flat R S› (f := G) (x := A) hs
    refine ⟨k,c,B,?_,hG⟩
    intro i
    simpa only [Algebra.smul_def,RingHom.algebraMap_toAlgebra,
      AlgHom.toRingHom_eq_coe,AlgHom.coe_toRingHom] using hc i
  · rintro ⟨k,c,B,hc,hG⟩
    have ht : Module.IsTrivialRelation G A := ⟨k,c,B,
      fun i => by simpa only [Algebra.smul_def,RingHom.algebraMap_toAlgebra,
        AlgHom.toRingHom_eq_coe,AlgHom.coe_toRingHom] using hc i,hG⟩
    simpa only [Algebra.smul_def,RingHom.algebraMap_toAlgebra,
      AlgHom.toRingHom_eq_coe,AlgHom.coe_toRingHom] using
      Module.sum_smul_eq_zero_of_isTrivialRelation ht
