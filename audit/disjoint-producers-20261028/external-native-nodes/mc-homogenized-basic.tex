\begin{lemma}[Elementary properties of the homogenized ideal]
\label{mc:homogenized-basic}
\uses{mc:homogenized-ideal}
\source{wlodarczyk05-hironaka,wlodarczyk09-analytic-resolution:page-0020}
\lean{TranscendentalNakamaye.Pending.HomogenizedIdeal.mem_of_isInSpanAt,TranscendentalNakamaye.Pending.HomogenizedIdeal.mem_congr,TranscendentalNakamaye.Pending.HomogenizedIdeal.mul_mem_mul,TranscendentalNakamaye.Pending.HomogenizedIdeal.pow_mem_idealPow,TranscendentalNakamaye.Pending.HomogenizedIdeal.iterDerivIdeal_succ_mem_congr,TranscendentalNakamaye.Pending.HomogenizedIdeal.le_homogenizedIdeal,TranscendentalNakamaye.Pending.HomogenizedIdeal.homogenizedIdeal_one}
\leanok
\(\mathfrak a\subset\mathcal H(\mathfrak a)\), and \(\mathcal H(\mathfrak a)=\mathfrak a\) if \(\mu=1\).
\end{lemma}
\begin{proof}
\leanok
The summand \(i=0\) is \(\mathfrak a\cdot T(\mathfrak a)^0=\mathfrak a\cdot\mathcal O=\mathfrak a\). For \(\mu=1\) it is the only summand, and the coherent ideal
\(\mathfrak a\) is closed under holomorphic combinations of its germs.
\end{proof}
