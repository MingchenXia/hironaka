\section{Actual formal blowup coordinates and generation of normalized original members}

Work in the genuine formal complex algebra $S=\mathbb C[[X_1,\ldots,X_n]]$.
Fix an actual coordinate subset $A\subset\{1,\ldots,n\}$, an index $e\in A$,
and actual complex constants $b_i$. Put $P=(X_i:i\in A)$ and $x=X_e$.
The actual formal chart substitution in
\cite[\S7, pp.~267--272, especially Lemma~7.22 and Corollary~7.24]{bierstone-milman97-invent-math}
is
\[
 \sigma(X_i)=\begin{cases}X_e&i=e,\\X_e(b_i+X_i)&i\in A\setminus\{e\},\\X_i&i\notin A.\end{cases}
\]
We construct the genuine formal algebra map and the actual exceptional
quotients, then use the already proved whole coordinate-power division.
The endpoint below concerns the ideal generated by normalized images of
EVERY original formal member. Identifying this ideal with the independently
specified all-power saturation, and deriving geometric compatibility of
standard vertices with an admissible center, are separate subsequent steps.

\begin{lemma}[The actual formal chart sends every whole center power to its literal exceptional power]
\label{hir:actual-formal-blowup-coordinate-power-image}
\uses{hir:actual-formal-coordinate-power-coefficient-kernel}
Each displayed coordinate image has actual zero constant coefficient,
so it internally defines a genuine complex formal substitution $\sigma:S\to S$.
Its images of the WHOLE original coordinate ideals satisfy
\[
 \sigma_*P=(x),\qquad \sigma_*(P^d)=(x^d)\quad(d\in\mathbb N).
\]

\leanok
\end{lemma}
\begin{proof}
Every displayed image contains an actual variable factor, and thus has
zero constant coefficient. The finite zero-constant family therefore
admits the genuine formal substitution, retaining the existing complex
scalar action. The variable formula is precisely the displayed formula.
Every selected coordinate image is divisible by the SAME actual $x$;
image of the true selected-variable span is therefore contained in $(x)$.
Conversely $e\in A$, and its actual image is $x$ itself, so $(x)$ is
contained in that whole ideal image. Actual image-of-power and the
principal-ideal power identity give the assertion for EVERY $d$, including
$d=0$.
\end{proof}

\begin{lemma}[Every genuine center-power member has a unique internally constructed exceptional quotient]
\label{hir:actual-formal-blowup-normalized-member}
\uses{hir:actual-formal-blowup-coordinate-power-image}
For every actual $F\in P^d$ there is a unique genuine formal series
$[F]_d\in S$ satisfying
\[
 x^d[F]_d=\sigma(F).
\]
The quotient is constructed from this actual membership and the preceding
whole-ideal image formula, with no exceptional-divisibility certificate supplied.

\end{lemma}
\begin{proof}
Actual membership supplies $\sigma(F)\in\sigma_*(P^d)=(x^d)$.
The true principal-ideal membership identity supplies an actual quotient,
which defines $[F]_d$. The actual variable $x$ has nonzero coefficient
at its unit exponent, hence is nonzero. The genuine formal power-series
ring over $\mathbb C$ is a domain, so $x^d\ne0$ and cancellation proves
uniqueness of the actual quotient.
\end{proof}

\begin{theorem}[Actual normalized original members are generated by the actual normalized vertices]
\label{hir:actual-formal-blowup-normalized-original-member-generation}
\uses{hir:actual-formal-blowup-normalized-member,hir:actual-formal-centre-power-whole-division-membership}
Fix any actual ideal $I\subset S$ and any monomial order. Let $f_a$ be
the internally chosen actual monic vertex series, and put
$w_a=\sum_{i\in A}a_i$. Assume explicitly the actual source conditions
$f_a\in P^{w_a}$ for every actual vertex $a$. Then for EVERY actual
$F\in I\cap P^d$, its genuine quotient belongs to the actual finite ideal
\[
 [F]_d\in\bigl([f_a]_{w_a}:a\text{ an actual vertex of }I\bigr).
\]
\lean{Hironaka.HironakaActualFormalBlowupMemberGeneration.actualNormalizedFormalVertex,Hironaka.HironakaActualFormalBlowupMemberGeneration.actualNormalizedFormalMember_mem_vertex_span}
\leanok
\end{theorem}
\begin{proof}
The actual whole coordinate-power division already constructs the SAME
whole quotients $Q_a$ with
\[
 F=\sum_a Q_af_a,\qquad Q_a\in P^{d-w_a}.
\]
Normalize these actual quotients using the preceding internally constructed
division: $\sigma(Q_a)=x^{d-w_a}[Q_a]_{d-w_a}$, and independently
$\sigma(f_a)=x^{w_a}[f_a]_{w_a}$. Since
$(d-w_a)+w_a=d+(w_a-d)$ in the natural numbers, actual multiplicativity
and the finite actual sum give
\[
 \sigma(F)=x^d\sum_a
 x^{w_a-d}[Q_a]_{d-w_a}[f_a]_{w_a}.
\]
The SAME actual left side is $x^d[F]_d$. Genuine domain cancellation
therefore identifies $[F]_d$ with the displayed finite sum, which belongs
to the claimed actual generated ideal. This includes both $w_a\le d$
and $w_a>d$, with the genuine truncated natural exponents, and includes
the zero ideal and the empty vertex family.
\end{proof}

\begin{theorem}[Both whole normalized-member ideals coincide]
\label{hir:actual-formal-blowup-whole-normalized-member-ideal}
\uses{hir:actual-formal-blowup-normalized-original-member-generation}
Under the SAME explicit source vertex-power hypotheses, the actual whole
ideal spanned by $[F]_d$ for EVERY $d\in\mathbb N$ and EVERY actual
$F\in I\cap P^d$ equals the actual finite ideal spanned by the
$[f_a]_{w_a}$.
\lean{Hironaka.HironakaActualFormalBlowupMemberGeneration.actualNormalizedOriginalMemberIdeal,Hironaka.HironakaActualFormalBlowupMemberGeneration.actualNormalizedOriginalMemberIdeal_eq_vertex_span}
\leanok
\end{theorem}
\begin{proof}
Every generator on the first side belongs to the second side by the
preceding genuine whole-member theorem; the actual ideal-span induction
gives that whole inclusion. Conversely each actual $f_a$ belongs to $I$
and, by the explicit source condition, to $P^{w_a}$. Its SAME normalized
series is therefore literally among the generators on the first side.
The actual ideal-span property gives the reverse whole inclusion.
No saturation equality or geometric admissibility hypothesis is silently
supplied by either ideal definition or either inclusion.
\end{proof}
