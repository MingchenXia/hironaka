\section{Whole strict ideals from actual centre-compatible monic families}

Let $S=\mathbb C[[x_1,\ldots,x_n]]$, let $J\subseteq S$ be an ideal,
and let $V$ be its true initial vertices for a fixed monomial order.
Let $(B_\alpha)_{\alpha\in V}$ be a monic vertex family: each member lies
in $J$, has coefficient $1$ at $\alpha$, and vanishes at every preceding
exponent. For a coordinate centre $A$, write $C_A$ for its formal ideal and
$w_\alpha=\sum_{i\in A}\alpha_i$. Fix the actual blowup chart with
exceptional coordinate $x_e$, $e\in A$, and chart point $b$; denote its
whole substitution by $\sigma$. Assume $B_\alpha\in C_A^{w_\alpha}$.
The unique normalized member of $F\in C_A^d$ is denoted
$N_d(F)=\sigma(F)/x_e^d$.

\begin{lemma}[Every normalized member is generated by the same normalized family]
\label{hir:actual-family-normalized-member-generation}
\uses{hir:actual-monic-family-centre-power-division,hir:actual-formal-blowup-normalized-original-member-generation}
\lean{Hironaka.HironakaActualFormalFamilyStrictGenerators.familyNormalizedFormalVertex,Hironaka.HironakaActualFormalFamilyStrictGenerators.familyNormalizedFormalMember_mem_span}
\leanok
For every $d\geq0$ and $F\in J\cap C_A^d$,
\[
 N_d(F)\in \bigl(N_{w_\alpha}(B_\alpha)\bigr)_{\alpha\in V}.
\]
\end{lemma}
\begin{proof}
Entire division by this same monic family gives
$F=\sum_\alpha Q_\alpha B_\alpha$ and the whole filtered division theorem
gives $Q_\alpha\in C_A^{d-w_\alpha}$, where subtraction is truncated
at zero. Put $q_\alpha=N_{d-w_\alpha}(Q_\alpha)$ and
$b_\alpha=N_{w_\alpha}(B_\alpha)$. Multiplicativity of actual substitution
and the defining equations for these normalized members give
\[
 \sigma(Q_\alpha B_\alpha)
 =x_e^{(d-w_\alpha)+w_\alpha}q_\alpha b_\alpha
 =x_e^d\bigl(x_e^{w_\alpha-d}q_\alpha b_\alpha\bigr).
\]
The exponent identity follows by separating $w_\alpha\leq d$ and
$d\leq w_\alpha$; it also holds at equality and at zero.
Apply $\sigma$ to the entire finite division identity. Cancellation of
the nonzero $x_e^d$ in the actual formal domain yields
$N_d(F)=\sum_\alpha x_e^{w_\alpha-d}q_\alpha b_\alpha$.
Every summand belongs to the stated ideal, proving the claim.
\end{proof}

\begin{theorem}[The whole source strict ideal is generated by this normalized family]
\label{hir:actual-formal-strict-ideal-monic-family-generators}
\uses{hir:actual-family-normalized-member-generation,hir:actual-formal-strict-ideal-normalized-member-identification}
\lean{Hironaka.HironakaActualFormalFamilyStrictGenerators.actualFormalStrictTransform_eq_family_span}
\leanok
The actual source formal strict transform ideal in this chart is
\[
 J^{\mathrm{str}}=\bigl(N_{w_\alpha}(B_\alpha)\bigr)_{\alpha\in V}.
\]
\end{theorem}
\begin{proof}
The whole source strict ideal equals the ideal generated by all
$N_d(F)$ with $F\in J\cap C_A^d$. The preceding lemma puts each such
generator in the stated family span, giving one inclusion. Conversely,
each $B_\alpha$ is a member of $J\cap C_A^{w_\alpha}$, so its normalized
member is one of the actual generators of that entire normalized-member
ideal. Taking spans proves the reverse inclusion. This argument uses
the actual coefficients and centre-power membership of the chosen monic
family; it does not require equality with any canonical reduced family.
\end{proof}
