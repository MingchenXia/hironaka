\begin{lemma}[The actual complement of a proper analytic subset is dense in its component]
\label{hir:general-two-component-complement-dense}
\uses{ana:closure-difference-analytic,ana:regular-component}
\lean{Hironaka.HironakaGeneralTwoAnalyticSuperlevels.closure_diff_eq_globalComponent_of_not_subset}
\leanok
For an actual global component $B$ of an analytic set and a genuine analytic
$Z$ not containing $B$, the closure of $B\setminus Z$ is exactly $B$.
\end{lemma}
\begin{proof}
The actual component $B$ is analytically irreducible and closed analytic.
The actual intersection $B\cap Z$ and actual closure
$\overline{B\setminus Z}$ are both analytic and contained in $B$.
Their union is exactly $B$: membership in $Z$ gives the first part, and
nonmembership gives the second by the real closure inclusion. Actual
analytic irreducibility makes one part all of $B$. The first alternative
would imply $B\subseteq Z$, contrary to the real hypothesis. Thus the
second is all of $B$. This argument includes every dimension and uses no
supplied density certificate.
\end{proof}

\begin{theorem}[Actual component selection and the zero-dimensional remainder]
\label{hir:general-two-component-superlevel-descent}
\uses{hir:general-two-component-complement-dense,hir:curve-predicates-locally-finite-union,hir:analytic-dimzero-subsets-and-constancy}
\lean{Hironaka.HironakaGeneralTwoAnalyticSuperlevels.eq_selected_globalComponents_union_exception,Hironaka.HironakaGeneralTwoAnalyticSuperlevels.isComplexAnalyticSubset_of_component_dictionary}
\leanok
Let $L$ be a genuinely closed subset of an analytic $A$, and let $Z$ be
analytic of dimension at most zero. If membership in $L$ agrees at any two
points of $B\setminus Z$ for every actual global component $B$ of $A$, then
\[
 L=\bigcup_{B\in\operatorname{globalComponents}(A),\ B\subseteq L}B
    \ \cup\ (L\cap Z),
\]
and this entire set is analytic.
\end{theorem}
\begin{proof}
A point of $L$ outside $Z$ lies on an actual global component $B$.
The membership dictionary puts every point of $B\setminus Z$ in $L$.
Since this complement is nonempty, $B$ is not contained in $Z$.
The preceding actual density producer and genuine closedness of $L$
fill the whole $B$ into $L$. A point of $L$ on $Z$ lies in $L\cap Z$.
The opposite inclusion is literal from selection and intersection.
The selected family is a genuinely locally finite subfamily of the actual
global components, so its actual analytic union is analytic. Every subset
of the actual analytic zero-dimensional $Z$ is closed smooth by its actual
singleton flattenings, so $L\cap Z$ is analytic. Taking the genuine finite
union proves whole-set analyticity. The component dictionary is a general
descent lemma input; the next producers establish it for the actual
invariant rather than assuming it.
\end{proof}
