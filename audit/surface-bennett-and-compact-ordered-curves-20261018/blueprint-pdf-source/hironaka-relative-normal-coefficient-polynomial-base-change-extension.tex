\section{Whole coefficient extension of polynomial quotients}

\begin{lemma}[The entire coefficient extension quotient is the actual tensor base change]
\label{hir:relative-normal-coefficient-polynomial-quotient-base-change}
\source{wlodarczyk05-hironaka}
\lean{Hironaka.HironakaRelativeNormalCoefficientPolynomialBaseChange.polynomialQuotientBaseChangeEquiv}
\leanok
Let $R\to S$ be a homomorphism of commutative rings and let
$H\subset R[Y_1,\ldots,Y_t]$ be an ideal. There is an $S$-algebra
isomorphism
\[
 S\otimes_R(R[Y]/H)\cong S[Y]/H_S,
\]
where $H_S$ is the whole ideal generated by the coefficient images of all
members of $H$.
\end{lemma}
\begin{proof}
Tensoring the whole quotient gives the quotient of $S\otimes_R R[Y]$ by
the ideal generated by $1\otimes H$. The polynomial tensor isomorphism
sends $s\otimes P$ to $sP_S$, where $P_S$ is coefficient extension.
It therefore sends the whole ideal generated by $1\otimes H$ exactly to
$H_S$. Passing to quotients gives the asserted isomorphism; both maps
preserve the scalar inclusion of $S$.
\end{proof}

\begin{corollary}[Whole polynomial quotient freeness survives actual coefficient extension]
\label{hir:relative-normal-coefficient-polynomial-free-base-change}
\source{wlodarczyk05-hironaka}
\lean{Hironaka.HironakaRelativeNormalCoefficientPolynomialBaseChange.moduleFree_coefficient_extended_quotient}
\leanok
\uses{hir:relative-normal-coefficient-polynomial-quotient-base-change}
If $R[Y]/H$ is free as an $R$-module, then $S[Y]/H_S$ is free as an
$S$-module.
\end{corollary}
\begin{proof}
Tensoring a free $R$-module with $S$ gives a free $S$-module, with the same
basis indices and extended basis vectors. Transfer this free module structure
through the preceding whole quotient isomorphism.
\end{proof}

\begin{theorem}[A single uniform denominator controls every coefficient-image quotient]
\label{hir:relative-normal-coefficient-uniform-denominator-free-images}
\source{wlodarczyk05-hironaka}
\lean{Hironaka.HironakaRelativeNormalCoefficientPolynomialBaseChange.moduleFree_coefficient_image_at_uniform_denominator}
\leanok
\uses{hir:relative-normal-coefficient-polynomial-free-base-change,hir:relative-normal-uniform-localized-whole-basis}
For an ideal $H\subset D[Y]$ over a domain $D$, let $b\ne0$ be its
uniform standard-monomial denominator. For every homomorphism $f:D\to S$
such that $f(b)$ is a unit, the entire quotient $S[Y]/H_S$ is free as an
$S$-module. The same $b$ works for every such coefficient homomorphism.
\end{theorem}
\begin{proof}
The unit condition extends $f$ to $D[b^{-1}]\to S$. The entire
coefficient-localized quotient is free by the uniform standard-monomial
theorem. Extend its coefficients through this actual localisation map and
apply the preceding free base-change theorem. The composition of the
localisation map with the extended map is the original $f$, so its whole
ideal image is exactly $H_S$. This proves the claimed free quotient using
the same internally determined denominator for every $f$.
\end{proof}
