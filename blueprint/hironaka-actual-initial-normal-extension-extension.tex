\label{hir:actual-polynomial-series-normal-rename-square}
\lean{Hironaka.HironakaActualInitialNormalExtension.polynomialSeries_normalRename_square,Hironaka.HironakaActualInitialNormalExtension.polynomialSeries_normalExtension_map}
\leanok

\[

indices ensure the actual series rename has finite fibres, also in

\end{proof}
