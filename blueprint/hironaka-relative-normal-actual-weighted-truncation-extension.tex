\begin{lemma}[The whole weighted sum of true tails is exactly the coordinate-weight submodule]
\label{hir:relative-normal-actual-weighted-tail-coordinate-dictionary}
\uses{hir:relative-normal-actual-power-lift-truncation-whole-kernel}
\lean{Hironaka.HironakaRelativeNormalActualWeightedTruncations.actualWeightedCoordinates,Hironaka.HironakaRelativeNormalActualWeightedTruncations.actualWeightedTailSum,Hironaka.HironakaRelativeNormalActualWeightedTruncations.actualDegreeRange_le_tail,Hironaka.HironakaRelativeNormalActualWeightedTruncations.actualWeightedTailSum_eq_coordinates}
\leanok
Let $D^{\geq q}$ be the original whole direct-sum tail. Then
\[
 W_s:=\sum_{q=0}^s\mathfrak m^{s-q}D^{\geq q}
 =\{v\in D:\ v_j\in\mathfrak m^{s-j}L_j\text{ for every }j\}.
\]
For $j\geq s$ the exponent is zero and imposes no restriction.
\end{lemma}
\begin{proof}
Project each true tail to degree $j$. If $j<q$ its image is zero;
if $j\geq q$ its image lies in all of $L_j$. Since
$\mathfrak m^{s-q}\subseteq\mathfrak m^{s-j}$ for $j\geq q$,
every projected summand has the displayed weight. Projection is
$B$-linear, so the whole finite sum does too.
Conversely write $v$ as the finite sum of its actual degree inclusions.
If $j<s$, the given coefficient membership and linearity of the degree
inclusion place this summand in $\mathfrak m^{s-j}D^{\geq j}$.
If $j\geq s$, this summand lies in $D^{\geq s}$, the $q=s$
summand with unit weight. Each term therefore lies in the true finite
weighted tail sum. Closure under addition proves the reverse inclusion.
\end{proof}
