\begin{lemma}[An actual ideal multiple of finite coefficient families is coordinatewise membership]
\label{hir:relative-normal-actual-free-quotient-coordinate-ideal}
\lean{Hironaka.HironakaRelativeNormalActualHSConvolution.actualCoefficientIdeal}
\leanok
For a commutative ring $B$, an actual ideal $K$ and a finite index $A$,
$K(B^A)$ consists exactly of families whose every coordinate belongs to $K$.
\end{lemma}
\begin{proof}
Each generator $b v$ with $b\in K$ has every coordinate in $K$, and
coordinatewise ideal membership is closed under finite sums and the
original $B$ action. This gives one inclusion. Conversely every such
family $v$ is the actual finite sum $\sum_{a\in A}v_a e_a$. Each genuine
coordinate coefficient $v_a$ belongs to $K$, so every summand belongs
to $K(B^A)$, proving the other inclusion. Empty indices give the zero
module on both sides.
\end{proof}

\begin{theorem}[An actual finite free quotient has genuine original quotient coefficients]

\begin{theorem}[Actual finite free layer quotients have the true rank factor]
\label{hir:relative-normal-actual-finite-free-quotient-dimension}
\uses{hir:relative-normal-actual-finite-free-quotient-dictionary}
\lean{Hironaka.HironakaRelativeNormalActualHSConvolution.actualFreeQuotientDimension,Hironaka.HironakaRelativeNormalActualHSConvolution.actualFreeQuotientFinite}
\leanok
If $B$ is a genuine nontrivial complex algebra and the actual quotient $B/K$ is finite-dimensional, then
the original complex dimension of $L/(KL)$ is
$\operatorname{rank}_B(L)\dim_{\mathbb C}(B/K)$.
\end{theorem}
\begin{proof}

have the same actual quotient dimension, so this sum is $|A|$ times that
dimension. The actual finite basis cardinality is the genuine $B$ module
rank, proving the formula. The same isomorphism internally supplies the
required finite-dimensionality of the original quotient.
\end{proof}

\begin{theorem}[Every actual weighted normal layer has its original tangent binomial dimension]
\label{hir:relative-normal-actual-weighted-layer-binomial-dimension}

original complex dimension of

\end{proof}
